\section{Introduction to Discrete Mathematics}\label{dm:section:1.idm}

\subsection{Arrangements and Permutations}

\subsubsection{Arrangements}

\definition{13.1.1.1}{An \underline{arrangement} is a sequence of items in which the order of the items in the sequence matters.}

\separate{7}\definition{13.1.1.2}{The \underline{length} of an arrangement is the number of items in the sequence.}

\separate{7}\standardproblem{1}{How many arrangements of length \(k\) can be made from elements of a set \(S\) of size \(|S|=n\)?}
{\(n^k\) as arrangement has \(k\) places, each of which can be occupied by any one of \(n\) entries.}

\separate{7}\example{13.1.1.1}{How many four-letter \say{words} can be made from the standard A-Z alphabet?

\separate{3}\textit{Solution}: \(26^4 = 456,796\) words.}

\separate{7}\standardproblem{2}{How many arrangements of length \(k\) with no repeats can be made from elements of a set \(S\) of size \(|S|=n\)?}
{\begin{equation} 
    n(n-1)(n-2)...(n-k+1) = \frac{n!}{(n-k)!},\qquad k \leq n    
\end{equation}
As we look at the available choices for each successive item in the list, the number of options is reduced by one each time.}

\separate{7}\definition{13.1.1.3}{The \underline{multiplication principle} states that, for an arrangement of length \(k\), if for \(i \in \{1,\;2,\;...,\;k\}\) there are \(n_i\) choices for the \(n^{th}\) item, regardless of previous choices, then the number of possible arrangements is \(n_1n_2...n_k\).}

\subsubsection{Permutations}

\definition{13.1.2.1}{A \underline{permutation} on a finite set \(S\) is an arrangement of length \(|S|\) that uses each item from \(S\) exactly once.}

\separate{7}\proposition{13.1.2.1}{If \(|S|=n\), then there are \(n!\) permutations on \(S\).}
{This is \hyperref[prob:2]{Standard Problem 2} with \(n=k\).}

\separate{7}\definition{13.1.2.2}{We generalise this concept as, for a non-negative integer \(r\), an \underline{\(r\)-permutation} on \(S\) is an arrangement of \(r\) distinct elements from \(S\). We write \(P(n,r)\) for the number of \(r\)-permutations on a set of size \(n\), which, by \hyperref[prob:2]{Standard Problem 2}, we have,
\begin{equation}
    P(n,r) = n(n-1)...(n-r+1) = \frac{n!}{(n-r)!},\qquad 0 \leq r \leq n
\end{equation}}

\subsubsection{Counting Examples}

\definition{13.1.3.1}{The \underline{addition principle} states that, for two disjoint, finite sets \(A\) and \(B\). Then,
\begin{equation}
    A \cap B = \emptyset \qquad |A \cup B| = |A| + |B|
\end{equation}
More generally, if \(A_1,\;A_2,\;...,\;A_n\) are pairwise disjoint, then
\begin{equation}
    |A_1 \cup A_2 \cup ... \cup A_n| = |A_1| + |A_2| + ... + |A_n|
\end{equation}}

\subsection{Combinations}

\definition{13.2.0.1}{A \underline{combination} from a set of objects is an unordered selection of objects of that set.}

\separate{7}\standardproblem{3}{How many combinations without repetition do we have of a given size \(k\) from a set of size \(n\)?}
{First consider that there are \(P(n,k) = \frac{n!}{(n-k)!}\) arrangements without repetition. These are ordered lists, sorting them into groups of lists using the same letters will give groups of size \(k!\). This means that the number of combinations is
\begin{equation}
    \frac{P(n,k)}{k!} = \frac{n!}{k!(n-k)!}
\end{equation}
Using the binomial coefficient notation, this means the number of combinations is \(\binom{n}{k}\).}

\separate{7}\example{13.2.0.1}{How many ways are there of selecting ten different letters of the alphabet?

\separate{3}\textit{Solution}: \(\binom{26}{10} = \frac{26!}{10!16!}\).}

\separate{7}\proposition{13.2.0.1}{\(\binom{n}{k} = \binom{n}{n-k}\)}{This can be seen from the formula for the binomial.}

\separate{7}\proposition{13.2.0.2}{Pascal's formula, for \(1 \leq k \leq (n-1),\;\binom{n}{k}=\binom{n-1}{k-1} + \binom{n-1}{k}\).}
{Proof excluded.}

\separate{7}\theorem{13.2.0.1}{The binomial theorem, if \(n\) is a positive integer and \(a,b \in \R\), then
\begin{equation}
    (a+b)^n = \sum^n_{k=0} \binom{n}{k} a^kb^{n-k}
\end{equation}}
{The expression \((a+b)(a+b)...(a+b)\) when multiplied out, gives various terms of the form \(a^kb^{n-k}\). For fixed \(k\), the number of terms \(a^kb^{n-k}\) is \(\binom{n}{k}\) since one chooses the \(k\) brackets in the product form from which to choose the factor \(a\) rather than \(b\).}

\standardproblem{4}{How many arrangements are there of a list of objects in which some objects are repeated?}

\newpage\noindent\example{13.3.1.1}{How many arrangements of REARRANGE are there, and of those, how many of the arrangements have at least two R's together?

\separate{3}\textit{Solution}: The letters to arrange are RRR AA EE G and N. Applying \hyperref[prf:thm:13.3.1.1]{Theorem 13.3.1.1} with \(n = 9\) and \(n_1 = 3,\;n_2 = 2,\;n_3 = 2,\;n_4 = 1,\;n_5 = 1\), we get
\begin{equation}
    \frac{9!}{3!2!2!} = 15,120
\end{equation}
We can then take all the cases where all the R's are separate. To do this, we arrange all the other letters, giving \(\frac{6!}{2!2!}\) arrangements. We then insert the R's into the available gaps, giving 7 possible spots for 3 R's, meaning the number of arrangements are
\begin{equation}
    \frac{9!}{3!2!2!} - \binom{7}{3}\left(\frac{6!}{2!2!}\right) = 8,820
\end{equation}}

\subsubsection{Combinations with Repetitions}

\standardproblem{5}{Given a set of \(n\) objects, how many combinations of size \(k\) can be chosen if objects may be selected more than once? (How many \(k\)-combinations with repetitions from \(n\) objects are there?)}
{To solve this problem, first consider}

\separate{7}\standardproblem{6}{How many \(k\)-combinations with repetitions from \(n\) objects are there in which each object is chosen at least once in the combination?}
{\theorem{13.3.2.1}{The number of \(k\)-combinations from \(n\) different objects with repetition and each object occuring at least once is \(\binom{k-1}{n-1}\) for \(k \geq n\).}
{This question clearly has a non-zero answer only if \(k \geq n\). Since order does not matter in our combinations, we represent each combination with its items in non-decreasing order. Now insert markers into the strings at the points where the numbers change. Because we have specified that each number appears at least once, there must be \((n-1)\) markers occupying \((k-1)\) possible positions in each of these lists.}}

\separate{7}\example{13.3.2.1}{Count the 5-combinations with repetitions from \(\{1,2,3\}\) in which each number appears at least once.

\separate{3}\textit{Solution}: Using the method above, we get the following combinations:
\begin{equation*} 1\,1\,1\,2\,3 \quad\leftrightarrow\quad 1\,1\,1\,|\,2\,|\,3 \end{equation*}
\begin{equation*} 1\,1\,2\,2\,3 \quad\leftrightarrow\quad 1\,1\,|\,2\,2\,|\,3 \end{equation*}
\begin{equation*} 1\,1\,2\,3\,3 \quad\leftrightarrow\quad 1\,1\,|\,2\,|\,3\,3 \end{equation*}
\begin{equation*} 1\,2\,2\,2\,3 \quad\leftrightarrow\quad 1\,|\,2\,2\,2\,|\,3 \end{equation*}
\begin{equation*} 1\,2\,2\,3\,3 \quad\leftrightarrow\quad 1\,|\,2\,2\,|\,3\,3 \end{equation*}
\begin{equation*} 1\,2\,3\,3\,3 \quad\leftrightarrow\quad 1\,|\,2\,|\,3\,3\,3 \end{equation*}
There are 4 possible gaps for 2 markers, so the number of ways of arranging the markers is \(\binom{4}{2} = 6\).}

\separate{7}We can adapt this solution to solve \hyperref[prob:5]{Standard Problem 5} by allowing the markers to also be placed next to one another, or outside of the item strings. Taking this into account, we get:

\separate{7}\theorem{13.3.2.2}{The number of \(k\)-combinations from \(n\) different objects with repetition is
\begin{equation}
    \binom{k+n-1}{n-1} = \binom{k+n-1}{k}
\end{equation}
}{As explained above.}

\begin{equation}
    \binom{-n}{k} = (-1)^k\binom{k+n-1}{k} = (-1)^k\binom{k+n-1}{n-1}

Which simplifies to give the claimed result.

\subsection{The Three Principles}

\subsubsection{Mathematical Induction}

\definition{14.1.1.1}{\underline{Mathematical induction} provides a method for establishing the truth of a sequence of statements, each of which is associated with a non-negative integer. In its simplest form, it gives efficient proofs of identities such as the binomial formula.}

\separate{7}\theorem{14.1.1.1}{The principle of mathematical induction states that: suppose that \(P(n)\) is a sequence of propositions labelled by integers \(n \geq n_0\). Suppose that
\begin{enumerate}[label=\alph*.]
    \item \(P(n_0)\) is true, and
    \item For any \(n \geq n_0\), the truth of \(P(n)\) implies the truth of \(P(n+1)\). 
\end{enumerate}
Then, for every \(n \geq n_0\), \(P(n)\) is true.}
{\(P(n_0)\) is true by assumption (a). Then by (b), \(P(n_0 + 1)\) is also true. Again, by (b), \(P(n_0 + 2)\) is also true, and so on. This chain of deductions eventually reaches any given \(n \geq n_0\), so \(P(n)\) is true for all \(n \geq n_0\).}

\definition{14.1.2.1}{The \underline{pidgeon-hole principle} states that if \(n\) envelopes are distributed among \(m\) pidgeon-holes, and \(n \geq m\), then at least one pidgeon-hole contains more than one envelope.}

\subsubsection{The Inclusion-Exclusion Principle}

\proposition{14.1.3.1}{We have
\begin{equation}
    |A_1 \cup A_2 \cup A_3| = |A_1| + |A_2| + |A_3| - |A_1 \cap A_2| - |A_1 \cap A_3| - |A_2 \cap A_3| + |A_1 \cap A_2 \cap A_3|
\end{equation}}
{Apply \(|A_1 \cup A_2| = |A_1| + |A_2| - |A_1 \cap A_2|\) repeatedly.}

\newpage\noindent\theorem{14.1.3.1}{The inclusion-exclusion formula states that, for a positive integer \(n\) and finite sets \(A_1,\;...,\;A_n\),
\begin{equation}
    |A_1 \cup A_2 \cup ... \cup A_n| = \sum^n_{k=1} \left((-1)^{k+1} \times \sum_{i_1<i_2<...<i_k} | A_{i_1} \cup A_{i_2} \cup ... \cup A_{i_k} | \right)
\end{equation}}
{A slightly tricky induction on \(n\), with the base case \(n=2\) given by \(|A_1 \cup A_2| = |A_1| + |A_2| - |A_1 \cap A_2|\).}

\separate{7}\definition{14.1.3.1}{A \underline{derangement} of the numbers \(\{1,\;2,\;...,\;n\}\) is a permutation in which no number is in its correct position.}

\separate{7}\proposition{14.1.3.2}{The number of derangements of \(1,\;2,\;...,\;n\) is given by
\begin{equation}
    d(n) = \sum_{k=0}^{n} (-1)^k \frac{n!}{k!}
\end{equation}}
{We subtract from \(n!\) the number of non-derangements to get
\begin{equation}
    d(n) = n!\sum_{k=0}^{n} (-1)^k \frac{1}{k!}
\end{equation}}
In particular, the proportion of permutations that are derangements is
\begin{equation}
    \frac{d(n)}{n!} = n!\sum_{k=0}^{n} (-1)^k \frac{1}{k!}
\end{equation}
This is the power series expansion of \(e^{-1}\), so
\begin{equation}
    \lim_{n \rightarrow \infty} \frac{d(n)}{n!} = \frac{1}{e}
\end{equation}

\subsection{Recurrence Relations}

\subsubsection{Basic Definitions}

\definition{14.2.1.1}{As defined by \hyperref[cit:recurrenceRelation]{(Perepelitsa, 2020, p.1)}, a \underline{recurrence relation} for the sequence \(\{a_n\}\) is an equation that expresses \(a_n\) in terms of one or more of the previous terms of the sequence for all integers \(n\) with \(n \geq n_0\), where \(n_0\) is a non-negative integer.}

\separate{7}\definition{14.2.1.2}{As defined by \hyperref[cit:recurrenceRelation]{(Perepelitsa, 2020, p.1)}, a sequence is called a \underline{solution} of a recurrence relation if its terms satisfy the recurrence relation.}

\separate{7}\definition{14.2.1.3}{As defined by \hyperref[cit:recurrenceRelation]{(Perepelitsa, 2020, p.1)}, the \underline{initial conditions} for a sequence specify the terms that precede the first term where the recurrence relation takes effect.}

\separate{7}Let \(\{a_n\}\) be a sequence indexed by integers \(n = 0,\;1,\;2,\;...\), we are given:
\begin{itemize}
    \item The initial conditions \(a_0,\;a_1,\;...,\;a_{r-1}\).
    \item The recurrence relation 
    \begin{equation}\label{eq:recRel}
        a_n = \sum_{i=1}^{r} c_ia_{n-i} = c_1a_{n-1} + ... + c_ra_{n-r}, \quad (n \geq r)
    \end{equation}
    where \(c_1,\;...,\;c_r\) are specified constants.
\end{itemize}

\newpage\noindent\definition{14.2.1.4}{The recurrence relation is \underline{linear} because no term involves more than a single factor of the \(a_i\), and it is \underline{homogeneous} because every term involves an \(a_i\).}

\separate{7}\definition{14.2.1.5}{We seek solutions to the recurrence relation \eqref{eq:recRel} of the form \(a_n = x^n, \; x \neq 0\). Substituting this into the relation, we get 
\begin{equation}
    x^n = \sum_{i=1}^{r} c_ix^{n-i}, \quad(n \geq r)
\end{equation}
Dividing through by \(x^{n-r}\), we get
\begin{equation}\label{eq:charEq}
    x^r - \sum_{i=1}^{r} c_ix^{r-i} = 0
\end{equation}
This is the \underline{characteristic equation}.}

\subsubsection{General Method for Solving Homogeneous Linear Recurrence Relations}

\begin{enumerate}
    \item Write down the characteristic equation of the recurrence relation as in \eqref{eq:charEq}.
    \item Find the equation's roots \(\alpha_1,\;\alpha_2,\;...,\;\alpha_r\).
    \item Then, for each \(i\), \(a_n = d(\alpha_i)^n\) satisfies the recurrence relation \eqref{eq:recRel} for any constant \(d\).
    \item By linearity, any combination of the \((\alpha_i)^n\) also solves the recurrence relation.
    \item Find the general solution to the recurrence relation. 
    \begin{itemize}
        \item If the roots are distinct, then the general solution to the recurrence relation is:
        \begin{equation}\label{eq:genSol}
            a_n = g(n) = d_1(\alpha_1)^n + d_2(\alpha_2)^n + ... + d_r(\alpha_r)^n
        \end{equation}
        for arbitrary constants \(d_1,\;d_2,\;...,\;d_r\).
        \item If \(\alpha\) is a root repeated \(k\) times, then in place of the expression \eqref{eq:genSol}, we use instead
        \begin{equation}
            a_n = g(n) = (e_1 + e_2n + ... + e_kn^{k-1})(\alpha_1)^n + d_{2}(\alpha_2)^n + ... + d_r(\alpha_r)^n
        \end{equation}
        for constants \(e_1,\;e_2,\;...,\;e_k\).
    \end{itemize} 
    \item Apply the initial conditions, giving \(r\) linear equations in the \(r\) unknown constants \(d_i\). Solve these to get a solution \(a_n\).
\end{enumerate}

\separate{7}\example{14.2.2.1}{Solve \(a_n = 3a_{n-1} + 4a_{n-2},\;(n \geq 2)\) where \(a_0 = 1\) and \(a_1 = 0\).

\separate{3}\textit{Solution}: The characteristic equation is 
\begin{equation}
    x^2 - 3x - 4 = 0 \qquad\rightarrow\qquad (x-4)(x+1) = 0 
\end{equation}
This has distinct roots \(x=4\) and \(x=-1\), so the general solution is 
\begin{equation}
    a_n = d_1(-1)^n + d_2(4)^n
\end{equation}
Using the initial conditions gives
\begin{equation*}
    1 = a_0 = d_1 + d_2 \qquad\text{and}\qquad 0 = a_1 = 4d_2 - d_1
\end{equation*}
Hence \(d_1 = \frac{4}{5}\) and \(d_2 = \frac{1}{5}\), so the solution is
\begin{equation}
    a_n = \frac{1}{5}\left(4^n\right) + \frac{4}{5}\left(-1^n\right)
\end{equation}}

\newpage\subsubsection{General Method for Solving Inhomogeneous Linear Recurrence Relations}

Redefine the recurrence relation \eqref{eq:recRel} to include an inhomogeneous part \(f(n)\),
\begin{equation}
    a_n = c_1a_{n-1} + ... + c_ra_{n-r} + f(n)
\end{equation} 

\begin{enumerate}
    \item Find the general solution to the homogeneous problem \(g(n) = a_n\).
    \item Find a specific particular solution \(p(n) = a_n\) to the full inhomogeneous problem using \hyperref[tab:Forms of the Particular Solution]{Table 1}.
    \begin{itemize}
        \item If \(f(n)\) contains \(q^n\) where \(q\) is a root of the characteristic equation, then \(q^n\) should be replaced by \(nq^{n-1}\).
        \item If 1 is a root of the characteristic equation, then \(f(n) = An^k\) should be considered as \(f(n) = An^k \cdot 1^n\) and so should be treated as above.
    \end{itemize}
    \item The general solution to the inhomogeneous problem is then the sum of these
    \begin{equation}
        a_n = g(n) + p(n)
    \end{equation}
    \item Apply the initial conditions to determine the unknown constants.
\end{enumerate}

\inserttable{c|c}{Forms of the Particular Solution}
{\textbf{f(n)} & \textbf{Form of Particular Solution \(p(n)\)}}
{\(A\) & \(B_0\) (constant) \\
\(An\) & \(B_0 + B_1n\) (linear) \\
\(An^2\) & \(B_0 + B_1n + B_2n^2\) (quadratic) \\
\(Aq^n\) & \(B_0q^n\) (exponential) \\
\(Anq^n\) & \((B_0 + B_1n)q^n\) (etc...)}

\noindent\example{14.2.3.1}{Solve \(a_n = a_{n-1} + 2a_{n-2} - 2n,\;(n \geq 2)\) where \(a_0 = a_1 = 0\).

\separate{3}\textit{Solution}: Solving the homogeneous recurrence relation, we get the characteristic equation
\begin{equation}
    x^2 - x - 2 = 0 \qquad\rightarrow\qquad (x-2)(x+1) = 0
\end{equation}
Which has distinct roots \(x=-1\) and \(x=2\), giving the general solution
\begin{equation}
    g(n) = d_1(-1)^n + d_2(2)^n
\end{equation}
The inhomogeneous part is \(f(n) = -2n\), so we seek a particular solution of the form
\begin{equation}
    p(n) = B_0 + B_1n
\end{equation}
Setting \(a_n = p(n)\) in the inhomogeneous recurrence relation gives
\begin{equation}
    B_1n + B_0 = \left(B_1(n-1) + B_0\right) + 2\left(B_1(n-2) + B_0\right) - 2n
\end{equation}
Which we solve to get \(B_1 = 1\) and \(B_0 = \frac{5}{2}\), meaning the particular solution is 
\begin{equation}
    p(n) = n + \frac{5}{2}
\end{equation}
This gives us the general solution to the inhomogeneous problem to be
\begin{equation}
    a_n = g(n) + p(n) = d_1(-1)^n + d_2(2)^n + n + \frac{5}{2}
\end{equation}
Substituting in the initial conditions, we get that 
\begin{equation}
    a_n = -\frac{1}{2}(-1)^n - 2^n + n + \frac{5}{2} 
\end{equation}
}

\subsection{Generating Functions}

\definition{14.3.0.1}{\underline{Generating functions} encode combinatorial information into an analytical form so we can solve counting problems. For example, the ordinary generating function for the sequence \(a_0,\;a_1,\;...\) is
\begin{equation}
    f(x) = \sum_{k=0}^{\infty} a_kx^k
\end{equation}}

\subsubsection{Counting with Generating Functions}

\example{14.3.1.1}{How many solutions are there to the equation
\begin{equation}\label{eq:ex.14.3.1.1}
    e_1 + e_2 + ... + e_n = k
\end{equation}
where each \(e_i \in \{0,1\}\) and \(n,k\) are given, positive integers?

\separate{3}\textit{Solution}: When constructing our generating function, each \(e_i\) is associated with the expression \((x^0 + x^1) = (1 + x)\), the generating function is then the \(n\)-fold product,
\begin{equation}
    f(x) = (1+x) \cdot (1+x) ... (1+x) = (1+x)^n
\end{equation}
If we expand the generating function as a power series, we get
\begin{equation}
    (1+x)^n = \sum_{k=0}^{n} \binom{n}{k} x^k
\end{equation}
Each term \(x^k\) corresponds to a choice of \(x^{e_1}\) from the first bracket, \(x^{e_2}\) from the second bracket, etc., in which the total power of \(x\) is \(e_1 + ... + e_n = k\), i.e., a solution to \eqref{eq:ex.14.3.1.1}.}

\separate{7}\lemma{14.3.1.1}{For \(n \geq 0, \;n \in \Z\), the generating function for the sequence 1,1,...1,0,0,... (first \((n+1)\) terms are 1) is
\begin{equation}\label{eq:genFunc 1, ..., 1, 0, ...}
    \sum_{k=0}^{n}x^k = \frac{1 - x^{n+1}}{1-x}
\end{equation}}
{This is the sum of a geometric series which can be proven directly in the form 
\begin{equation}
    (1-x)\sum_{k=0}^{n} x^k = \sum_{k=0}^{n} \left(x^k - x^{k-1}\right) = 1 - x^{n+1}
\end{equation}
since the sum telescopes.}

\separate{7}\lemma{14.3.1.2}{For \(n \geq 0, \;n \in \Z\), the generating function for the sequence 1,1,1,... is
\begin{equation}\label{eq:genFunc 1, ..., 1, ...}
    \sum_{k=0}^{n}x^k = \frac{1}{1-x}
\end{equation}}
{The sum to infinity makes analytical sense only for \(|x| < 1\), in which case we can take \(n \rightarrow \infty\) in \eqref{eq:genFunc 1, ..., 1, 0, ...} to get \eqref{eq:genFunc 1, ..., 1, ...}.}

\newpage\noindent\lemma{14.3.1.3}{For \(n \geq 0, \;n \in \Z\), the generating function for the sequence \(\binom{n+k-1}{k},\;(k \geq 0)\) is
\begin{equation}\label{eq:genFunc binomial}
    \sum_{k=0}^{n}\binom{n+k-1}{k}x^k = \frac{1}{(1-x)^n}
\end{equation}}
{Recall from the extended form of the binomial theorem that
\begin{equation}
    (1-x)^{-n} = \sum_{k=0}^{\infty} \binom{n+k-1}{k} x^k
\end{equation}
As claimed.}

\separate{7}\lemma{14.3.1.4}{For \(n \geq 0, \;n \in \Z\), if \(f(x) = \sum_{k=0}^{\infty} a_kx^k\) and \(g(x) = \sum_{k=0}^{\infty} b_kx^k\) are the generating functions of sequences \(a_k\) and \(b_k\) respectively, then
\begin{equation}\label{eq:genFunc multiply}
    f(x)g(x) = \sum_{k=0}^{\infty} c_kx^k
\end{equation}
is the generating function of the sequence \(c_k = \sum_{l = 0}^{k} a_lb_{k-l}\).}
{This is just collecting terms.}

\separate{7}\example{14.3.1.2}{Find the coefficient of \(x^17\) in the expression \(f(x) = \left(1+x+x^2+...+x^{10}\right)^5\).

\separate{3}\textit{Solution}: We use \hyperref[lem:14.3.1.2]{Lemma 14.3.1.2} and \hyperref[lem:14.3.1.4]{Lemma 14.3.1.4} to write
\begin{equation}
    f(x) = \left(\frac{1-x^{11}}{1-x}\right)^5 = (1-x)^{-5}(1-x^{11})^5
\end{equation}
Considering separately the two terms in the product, we have
\begin{equation}
    (1-x^{11})^5 = \sum_{i=0}^{5} \binom{5}{i} (-1)^ix^{11i}
\end{equation}
And
\begin{equation}
    (1-x)^{-5} = \sum_{j=0}^{\infty} \binom{4+j}{4} x^j
\end{equation}
The terms in the product 
\begin{equation}
    \left(\sum_{i=0}^{5} \binom{5}{i} (-1)^ix^{11i}\right) \left(\sum_{j=0}^{\infty} \binom{4+j}{4} x^j\right)
\end{equation}
that contribute \(x^17\) are \((i,j) = (0,17),\;(1,6)\), so the required coefficient is 
\begin{equation}
    \binom{5}{0}\binom{21}{4} - \binom{5}{1}\binom{10}{4} = 5,985 - 1,050 = 4,935
\end{equation}}

\newpage\subsubsection{Recurrence Relations and Generating Functions}

\example{14.3.2.1}{Solve for \(a_n\) satisfying the recurrence relation \(a_n = a_{n-1} + 2a_{n-2},\;(n \geq 2)\) with initial conditions \(a_0=1\) and \(a_1 = -1\).

\separate{3}\textit{Idea}: Write the generating function for \(a_n\) as \(f(x) = \sum_{n=0}^{\infty}a_nx^n\). In this sum, substitute in for \(a_n\) using the recurrence relation for those values of \(n\) for which it is valid, then tidy up the remaining terms using the initial conditions.

\separate{3}\textit{Solution}: In this case, the recurrence relation allows us to replace \(a_n\) by \(a_{n-1} + 2a_{n-2}\) for all \(n \geq 2\), so we can write
\begin{equation}
    f(x) = 1 - x + \sum_{n=2}^{\infty}a_{n-1}x^n + 2\sum_{n=2}^{\infty}a_{n-2}x^n
\end{equation}
We want to relate the two sums to \(f(x)\) so we take out factors of \(x\) so that the index \(a_k\) matches the power \(x^k\) each time
\begin{equation}
    f(x) = 1 - x + x\sum_{n=2}^{\infty}a_{n-1}x^{n-1} + 2x^2\sum_{n=2}^{\infty}a_{n-2}x^{n-2}
\end{equation}
We can then re-label the summation indices to get \(a_n\) everywhere
\begin{equation}
    f(x) = 1 - x + x\sum_{n=1}^{\infty}a_{n}x^{n} + 2x^2\sum_{n=0}^{\infty}a_{n}x^{n}
\end{equation}
We can see that the last term is \(\sum_{n=0}^{\infty}a_{n}x^{n} = f(x)\) by definition and the first summation is missing the first term, so we can rewrite the equation as
\begin{equation}
    f(x) = 1 - x + x\left(f(x) - 1\right) + 2x^2f(x)
\end{equation}
Rearranging for \(f(x)\), we get that
\begin{equation}
    f(x) = \frac{1}{1+x} = 1 - x + x^2 - x^3 ...
\end{equation}
So our solution is \(a_n = 1\) if \(n\) is even, and \(a_n = -1\) if \(n\) is odd.}

\section{Graph Theory}\label{dm:section:3.gt}

\subsection{What is a Graph?}

\subsubsection{Basic Definitions}

\definition{15.1.1.1}{A \underline{simple graph} \(G\) is a pair of sets \(G = (V,E)\) where each element of \(E\) is a 2-element subset of \(V\).}

\separate{7}\definition{15.1.1.2}{Given a graph \(G=(V,E)\), if \(|V|\) and \(|E|\) are finite, then \(G\) is a \underline{finite graph}.}

\separate{7}\definition{15.1.1.3}{Two vertices are \underline{adjacent} if there is an edge between them, that is, \(u,v \in V\) are adjacent if \(\{u,v\} \in E\).}

\separate{7}\definition{15.1.1.4}{An edge is \underline{incident} to a vertex if that vertex is one of the endpoints of the edge.}

\subsubsection{Simple Graphs}

\definition{15.1.2.1}{The \underline{empty/null graph} on \(m\) vertices is called \(N_m\) where \(|V| = m\) but \(|E| = 0\).}

\separate{7}\definition{15.1.2.2}{The \underline{complete graph} on \(n\) is called \(K_n\). Given \(|V|=n\), we include an edge between every pair of vertices, so \(|E| = \binom{n}{2}\).}

\separate{7}\definition{15.1.2.3}{The \underline{cycle graph} on \(n\) vertices is called \(C_n\). Given \(|V|=n\), the cycle graph is constructed by listing the vertices in some order \(v_1,\;v_2,\;...,\;v_n\) and then adding the cycle of \(n\) edges \(v_1v_2,\;...,\;v_{n-1}v_n,\;v_n,v_1\).}

\separate{7}\definition{15.1.2.4}{The \underline{star graph} on \(n\) vertices is called \(S_n\). Of the \(|V|=n\) vertices, \((n-1)\) are leaves with a single edge to a central hub vertex.}

\separate{7}\definition{15.1.2.5}{The \underline{wheel graph} on \(n\) vertices is called \(W_n\). It can be constructed from the star graph \(S_n\) by adding a cycle around the leaves of \(S_n\).}

\separate{7}\definition{15.1.2.6}{The \underline{\(n\)-cube} has \(2^n\) vertices and is denoted \(Q_n\).}

\subsubsection{Handshaking}

\lemma{15.1.3.1}{For any graph \(G=(V,E)\), we have \(\sum_{v \in V} = 2|E|\).}{Each edge contributes 2 to the sum of degrees, one to each vertex it is incident to.}

\separate{7}\theorem{15.1.3.1}{In any graph, the number of odd-degree vertices is even.}
{Let \(|V| = n\) and \(|E| = m\), and let the number of vertices of odd degree be \(k\). By \hyperref[lem:15.1.3.1]{Lemma 15.1.3.1}, we have that the sum of degrees is even, and furthermore the sum of the degrees of the \((n-k)\) vertices of even degree is a sum of even numbers, and thus is even. 

Therefore, the sum of the degrees of the remaining \(k\) vertices of odd degree must also be even. As a sum of odd numbers, there must be an even number of terms for the sum to be even, and hence there must be an even number of vertices of odd degree.}

\newpage\subsubsection{Graph Isomorphism, Complements, and Subgraphs}

\definition{15.1.4.1}{Two simple graphs \(G_1\) and \(G_2\) are \underline{isomorphic}, \(G_1 \cong G_2\), if there exists a bijective function mapping the vertices from \(G_1\) to the vertices of \(G_2\) which preserves adjacency.}

\separate{7}\definition{15.1.4.2}{Let \(G=(V,E)\) be a simple graph. Then the \underline{complement} of \(G\) is \(\overline{G} = (V,F)\) where \(F = \{uv : u,v \in V \text{ and } uv \notin E\}\).}

\separate{7}\definition{15.1.4.3}{Given a graph \(G=(V,E)\) and a graph \(G_1 = (V_1, E_1)\) such that \(V_1 \subseteq V\) and \(E_1 \subseteq E\), then \(G_1\) is a \underline{subgraph} of \(G\).}

\subsection{Eulerian Graphs}

\subsubsection{Travelling in Graphs}

\definition{15.2.1.1}{Given a graph \(G=(V,E)\), a \underline{walk} in \(G\) is a sequence of vertices \(v_1,\;v_2,\;...,v_k\) from \(V\) together with edges \(e_1,\;e_2,\;...,\;e_{k-1}\) in \(E\) such that \(e_i = \{v_i,\;v_{i+1}\}\) for all \(1 \leq i \leq (k-1)\) where the \underline{length} of the walk is the number of edges. If \(v_1=v_k\), the walk is \underline{closed}.}

\separate{7}\definition{15.2.1.2}{A \underline{trail} is a sequence of vertices \(v_1,\;v_2,\;...,v_k\) from \(V\) together with distinct edges \(e_1,\;e_2,\;...,\;e_{k-1}\) in \(E\) such that \(e_i = v_iv_{i+1}\). That is, a trail is a walk with no edge repeated. A \underline{circuit} is a closed trail.}

\separate{7}\definition{15.2.1.3}{A \underline{path} is a series of distinct vertices, i.e., a walk which never revisits any previous vertex and a \underline{cycle} is a closed path.}

\separate{7}\definition{15.2.1.4}{A \underline{Hamilton cycle} in a graph \(G\) is a cycle which visits every vertex of \(G\). A graph which contains a Hamilton cycle is called \underline{Hamiltonian}.} 

\separate{7}\lemma{15.2.1.1}{Given vertices \(u \neq w\) in a graph \(G\), if there is a walk from \(u\) to \(w\) in \(G\), then there is a path from \(u\) to \(w\) in \(G\).}
{Let \(W\) be a walk in \(G\) consisting of the vertices \(v_i,\;1 \leq i \leq k,\;v_1 = u,\;v_2 = w\) and edges \(e_i,\; 1 \leq i \leq (k-1)\). If \(v_1,\;...,\;v_k\) are all distinct, \(W\) is a path. 

If not, let \(x=v_j\) be the first occurence in \(W\) of a vertex which appears more than once, and let \(v_{j'}\) be the latest occurence of that vertex in \(W\). Let \(W'\) be the walk formed from \(W\) by removing the vertices \(v_{j+1},\;...,\;v_{j'}\) and the edges \(e_j,\;...,\;e_{j-1}\). Then \(x\) appears only once in \(W'\) and so the number of repeated vertices in \(W'\) is strictly smaller than \(W\). 

We can then repeat this process starting with the walk \(W'\). As each time we repeat, the number of vertices appearing more than once strictly decreases, and eventually we must have a walk containing no repeated vertices, that is, a path.}

\separate{7}\lemma{15.2.1.2}{If every vertex in a finite graph \(G\) has degree \(> 1\), then \(G\) contains a cycle.}
{Let \(|V| = n\). Starting from any vertex \(u\) in \(G\), then as \(d(u) > 1\), there is some edge \(e=uv\) incident to \(u\). Let \(u=v_1,\;v=v_2\). We construct a cycle in \(G\) as follows: at \(v_2\), as \(d(v_2) > 1\), there must be some edge \(e_2 \neq e\) incident to \(v_2\). There are two possibilities:
\begin{enumerate}[label=(\alph*)]
    \item There is an edge from \(v_i\) to some already visited \(v_k\) with \(1 \leq k \leq i\).
    \item We do not have (a), and there is some edge \(e_i = \{v_i,v_{i+1}\}\) to some unvisited new vertex \(v_{i+1}\).
\end{enumerate}
But at some point \(i \leq n\), we must have situation (a) occuring - we cannot keep finding new vertices. Then the path \(v_k,\;v_{k+1},\;...,\;v_i,\;v_k\) is a cycle in \(G\) as required.}

\newpage\subsubsection{Connectedness}

\definition{15.2.2.1}{A graph \(G\) is \underline{connected} if, given any two vertices, there is a path between them.}

\separate{7}\definition{15.2.2.2}{A \underline{connected component} of \(G=(V,E)\) is a subgraph \(G_1 = (V_1,E_1)\) of \(G\) such that:
\begin{itemize}
    \item \(G_1\) is connected.
    \item For any \(u \in V_1\) and any \(v \in V\) such that there is a path from \(u\) to \(v\) in \(G\), then \(v \in V_1\).
    \item For every edge \(e=uv \in E\) where \(u,v \in V_1\), then \(e \in E_1\).
\end{itemize}}

\subsubsection{Euler's Theorem}

\definition{15.2.3.1}{An \underline{Euler circuit} in a graph \(G\) is a circuit which uses each edge of \(G\) exactly once. A graph with an Euler circuit is called \underline{Eulerian}.}

\separate{7}\theorem{15.2.3.1}{Let \(G = (V,E)\) be a finite connected graph with \(|V| > 1\). Then \(G\) has an Euler circuit \(\iff\) all vertices of \(G\) have even degree.}
{We have already seen why \(\implies\), to prove \(\impliedby\), we require the following Lemma:

\separate{7}\lemma{15.2.3.1}{If a graph \(G=(V,E)\) is finite and connected, with \(|V| > 1\) and all vertices of even degree, then \(G\) is a finite union of edge-disjoint cycles, that is, there are cycles \(C_1,\;C_2,\;...,\;C_n\) such that the union of the vertices of these cycles is equal to \(V\) and the union of edges is equal to \(E\).}
{Since \(G\) is connected, it has no isolated vertices, so all vertices have degree \(\geq 2\), by \hyperref[lem:15.2.1.2]{Lemma 15.2.1.2}, there is a cycle \(C_1\) in \(G\). We form a new graph from \(G\) by removing the all edges from \(C_1\). The vertices in \(C_1\) have their degrees decreased by 2 and then discard any which have become isolated. We can then repeat the process until the whole graph has been decomposed into cycles.}

\noindent\textit{Proof of Theorem 15.2.3.1 \((\impliedby)\)}: Let \(G\) be finite, connected, and with all degrees even. Then, by \hyperref[lem:15.2.3.1]{Lemma 15.2.3.1}, \(G\) is a finite union of \(n\) edge-disjoint cycles. We shall show, by induction on \(n\), that these can be combined into a single Eulerian circuit.
\begin{itemize}
    \item Base Case: If \(n=1\), the one cycle is the Euler circuit.
    \item Induction Hypothesis: If \(G\) is made up of \(\geq (n-1)\) edge-disjoint cycles, it has an Euler circuit.  
    \item Inductive Step: Now consider a graph \(G\) made up of \(n\) edge-disjoint cycles,
    \begin{equation}
        G = C_1 \cup C_2 \cup ... \cup C_{n-1} \cup C_n
    \end{equation}
    And let 
    \begin{equation}
        H = C_1 \cup C_2 \cup ... \cup C_{n-1}
    \end{equation}
    As in the proof of \hyperref[lem:15.2.3.1]{Lemma 15.2.3.1}, \(H\) has all degrees even. It may not be connected, so consider its connected components, \(H_1,\;H_2,\;...,\;H_k\). Since each cycle \(C_i\) lies within only one connected component \(H_j\), we know that there are at most \((n-1)\) connected components. 
    
    Furthermore, since the union \(C_1 \cup C_2 \cup ... \cup C_{n-1}\) makes up the whole of \(H\), each \(H_j\) must be made up by \(\leq (n-1)\) edge-disjoint cycles. So, by the Induction Hypothesis, each connected component \(H_j\) has an Euler circuit \(E_j\).

    But since \(G\) was connected, \(C_n\) must visit each \(H_j\), so we can make an Euler circuit for \(G\) by following \(C_n\), but when we first visit each \(H_j\), follow \(E_j\) all the way round before continuing.
\end{itemize}}
