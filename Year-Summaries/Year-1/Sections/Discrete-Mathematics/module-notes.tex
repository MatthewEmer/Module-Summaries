\subsection{Arrangements and Permutations}

\subsubsection{Arrangements}

\definition{13.1.1.1}{An \underline{arrangement} is a sequence of items in which the order of the items in the sequence matters.}

\separate{7}\definition{13.1.1.2}{The \underline{length} of an arrangement is the number of items in the sequence.}

\separate{7}\standardproblem{1}{How many arrangements of length \(k\) can be made from elements of a set \(S\) of size \(|S|=n\)?}
{\(n^k\) as arrangement has \(k\) places, each of which can be occupied by any one of \(n\) entries.}

\separate{7}\example{13.1.1.1}{How many four-letter \say{words} can be made from the standard A-Z alphabet?

\separate{3}\textit{Solution}: \(26^4 = 456,796\) words.}

\separate{7}\standardproblem{2}{How many arrangements of length \(k\) with no repeats can be made from elements of a set \(S\) of size \(|S|=n\)?}
{\begin{equation} 
    n(n-1)(n-2)...(n-k+1) = \frac{n!}{(n-k)!},\qquad k \leq n    
\end{equation}
As we look at the available choices for each successive item in the list, the number of options is reduced by one each time.}

\separate{7}\definition{13.1.1.3}{The \underline{multiplication principle} states that, for an arrangement of length \(k\), if for \(i \in \{1,\;2,\;...,\;k\}\) there are \(n_i\) choices for the \(n^{th}\) item, regardless of previous choices, then the number of possible arrangements is \(n_1n_2...n_k\).}

\subsubsection{Permutations}

\definition{13.1.2.1}{A \underline{permutation} on a finite set \(S\) is an arrangement of length \(|S|\) that uses each item from \(S\) exactly once.}

\separate{7}\proposition{13.1.2.1}{If \(|S|=n\), then there are \(n!\) permutations on \(S\).}
{This is \hyperref[prob:2]{Standard Problem 2} with \(n=k\).}

\separate{7}\definition{13.1.2.2}{We generalise this concept as, for a non-negative integer \(r\), an \underline{\(r\)-permutation} on \(S\) is an arrangement of \(r\) distinct elements from \(S\). We write \(P(n,r)\) for the number of \(r\)-permutations on a set of size \(n\), which, by \hyperref[prob:2]{Standard Problem 2}, we have,
\begin{equation}
    P(n,r) = n(n-1)...(n-r+1) = \frac{n!}{(n-r)!},\qquad 0 \leq r \leq n
\end{equation}}

\subsubsection{Counting Examples}

\definition{13.1.3.1}{The \underline{addition principle} states that, for two disjoint, finite sets \(A\) and \(B\). Then,
\begin{equation}
    A \cap B = \emptyset \qquad |A \cup B| = |A| + |B|
\end{equation}
More generally, if \(A_1,\;A_2,\;...,\;A_n\) are pairwise disjoint, then
\begin{equation}
    |A_1 \cup A_2 \cup ... \cup A_n| = |A_1| + |A_2| + ... + |A_n|
\end{equation}}

\subsection{Combinations}

\definition{13.2.0.1}{A \underline{combination} from a set of objects is an unordered selection of objects of that set.}

\separate{7}\standardproblem{3}{How many combinations without repetition do we have of a given size \(k\) from a set of size \(n\)?}
{First consider that there are \(P(n,k) = \frac{n!}{(n-k)!}\) arrangements without repetition. These are ordered lists, sorting them into groups of lists using the same letters will give groups of size \(k!\). This means that the number of combinations is
\begin{equation}
    \frac{P(n,k)}{k!} = \frac{n!}{k!(n-k)!}
\end{equation}
Using the binomial coefficient notation, this means the number of combinations is \(\binom{n}{k}\).}

\separate{7}\example{13.2.0.1}{How many ways are there of selecting ten different letters of the alphabet?

\separate{3}\textit{Solution}: \(\binom{26}{10} = \frac{26!}{10!16!}\).}

\separate{7}\proposition{13.2.0.1}{\(\binom{n}{k} = \binom{n}{n-k}\)}{This can be seen from the formula for the binomial.}

\separate{7}\proposition{13.2.0.2}{Pascal's formula, for \(1 \leq k \leq (n-1),\;\binom{n}{k}=\binom{n-1}{k-1} + \binom{n-1}{k}\).}
{Proof excluded.}

\separate{7}\theorem{13.2.0.1}{The binomial theorem, if \(n\) is a positive integer and \(a,b \in \R\), then
\begin{equation}
    (a+b)^n = \sum^n_{k=0} \binom{n}{k} a^kb^{n-k}
\end{equation}}
{The expression \((a+b)(a+b)...(a+b)\) when multiplied out, gives various terms of the form \(a^kb^{n-k}\). For fixed \(k\), the number of terms \(a^kb^{n-k}\) is \(\binom{n}{k}\) since one chooses the \(k\) brackets in the product form from which to choose the factor \(a\) rather than \(b\).}

\subsubsection{Arrangements with Repetitions}

\standardproblem{4}{How many arrangements are there of a list of objects in which some objects are repeated?}
{\theorem{13.3.1.1}{Given a list of \(n\) objects of \(r\) different types, in which objects of type \(i\) appear \(n_i\) times, the number of arrangements is
\begin{equation}
    P(n;\,n_1,\,n_2,\,...,\,n_r) = \frac{n!}{n_1!n_2!...n_r!}
\end{equation}}{Proof omitted.}}

\newpage\noindent\example{13.3.1.1}{How many arrangements of REARRANGE are there, and of those, how many of the arrangements have at least two R's together?

\separate{3}\textit{Solution}: The letters to arrange are RRR AA EE G and N. Applying \hyperref[prf:thm:13.3.1.1]{Theorem 13.3.1.1} with \(n = 9\) and \(n_1 = 3,\;n_2 = 2,\;n_3 = 2,\;n_4 = 1,\;n_5 = 1\), we get
\begin{equation}
    \frac{9!}{3!2!2!} = 15,120
\end{equation}
We can then take all the cases where all the R's are separate. To do this, we arrange all the other letters, giving \(\frac{6!}{2!2!}\) arrangements. We then insert the R's into the available gaps, giving 7 possible spots for 3 R's, meaning the number of arrangements are
\begin{equation}
    \frac{9!}{3!2!2!} - \binom{7}{3}\left(\frac{6!}{2!2!}\right) = 8,820
\end{equation}}

\subsubsection{Combinations with Repetitions}

\standardproblem{5}{Given a set of \(n\) objects, how many combinations of size \(k\) can be chosen if objects may be selected more than once? (How many \(k\)-combinations with repetitions from \(n\) objects are there?)}
{To solve this problem, first consider}

\separate{7}\standardproblem{6}{How many \(k\)-combinations with repetitions from \(n\) objects are there in which each object is chosen at least once in the combination?}
{\theorem{13.3.2.1}{The number of \(k\)-combinations from \(n\) different objects with repetition and each object occuring at least once is \(\binom{k-1}{n-1}\) for \(k \geq n\).}
{This question clearly has a non-zero answer only if \(k \geq n\). Since order does not matter in our combinations, we represent each combination with its items in non-decreasing order. Now insert markers into the strings at the points where the numbers change. Because we have specified that each number appears at least once, there must be \((n-1)\) markers occupying \((k-1)\) possible positions in each of these lists.}}

\separate{7}\example{13.3.2.1}{Count the 5-combinations with repetitions from \(\{1,2,3\}\) in which each number appears at least once.

\separate{3}\textit{Solution}: Using the method above, we get the following combinations:
\begin{equation*} 1\,1\,1\,2\,3 \quad\leftrightarrow\quad 1\,1\,1\,|\,2\,|\,3 \end{equation*}
\begin{equation*} 1\,1\,2\,2\,3 \quad\leftrightarrow\quad 1\,1\,|\,2\,2\,|\,3 \end{equation*}
\begin{equation*} 1\,1\,2\,3\,3 \quad\leftrightarrow\quad 1\,1\,|\,2\,|\,3\,3 \end{equation*}
\begin{equation*} 1\,2\,2\,2\,3 \quad\leftrightarrow\quad 1\,|\,2\,2\,2\,|\,3 \end{equation*}
\begin{equation*} 1\,2\,2\,3\,3 \quad\leftrightarrow\quad 1\,|\,2\,2\,|\,3\,3 \end{equation*}
\begin{equation*} 1\,2\,3\,3\,3 \quad\leftrightarrow\quad 1\,|\,2\,|\,3\,3\,3 \end{equation*}
There are 4 possible gaps for 2 markers, so the number of ways of arranging the markers is \(\binom{4}{2} = 6\).}

\separate{7}We can adapt this solution to solve \hyperref[prob:5]{Standard Problem 5} by allowing the markers to also be placed next to one another, or outside of the item strings. Taking this into account, we get:

\separate{7}\theorem{13.3.2.2}{The number of \(k\)-combinations from \(n\) different objects with repetition is
\begin{equation}
    \binom{k+n-1}{n-1} = \binom{k+n-1}{k}
\end{equation}
}{As explained above.}

\begin{equation}
    \binom{-n}{k} = (-1)^k\binom{k+n-1}{k} = (-1)^k\binom{k+n-1}{n-1}

Factoring out a factor of -1,
\begin{equation}

Which simplifies to give the claimed result.

\subsubsection{Mathematical Induction}

\definition{14.1.1.1}{\underline{Mathematical induction} provides a method for establishing the truth of a sequence of statements, each of which is associated with a non-negative integer. In its simplest form, it gives efficient proofs of identities such as the binomial formula.}

\separate{7}\theorem{14.1.1.1}{The principle of mathematical induction states that: suppose that \(P(n)\) is a sequence of propositions labelled by integers \(n \geq n_0\). Suppose that
\begin{enumerate}[label=\alph*.]
    \item \(P(n_0)\) is true, and
    \item For any \(n \geq n_0\), the truth of \(P(n)\) implies the truth of \(P(n+1)\). 
\end{enumerate}
Then, for every \(n \geq n_0\), \(P(n)\) is true.}
{\(P(n_0)\) is true by assumption (a). Then by (b), \(P(n_0 + 1)\) is also true. Again, by (b), \(P(n_0 + 2)\) is also true, and so on. This chain of deductions eventually reaches any given \(n \geq n_0\), so \(P(n)\) is true for all \(n \geq n_0\).}

\subsubsection{The Inclusion-Exclusion Principle}

\proposition{14.1.3.1}{We have
\begin{equation}
    |A_1 \cup A_2 \cup A_3| = |A_1| + |A_2| + |A_3| - |A_1 \cap A_2| - |A_1 \cap A_3| - |A_2 \cap A_3| + |A_1 \cap A_2 \cap A_3|
\end{equation}}
{Apply \(|A_1 \cup A_2| = |A_1| + |A_2| - |A_1 \cap A_2|\) repeatedly.}

\newpage\noindent\theorem{14.1.3.1}{The inclusion-exclusion formula states that, for a positive integer \(n\) and finite sets \(A_1,\;...,\;A_n\),
\begin{equation}
    |A_1 \cup A_2 \cup ... \cup A_n| = \sum^n_{k=1} \left((-1)^{k+1} \times \sum_{i_1<i_2<...<i_k} | A_{i_1} \cup A_{i_2} \cup ... \cup A_{i_k} | \right)
\end{equation}}
{A slightly tricky induction on \(n\), with the base case \(n=2\) given by \(|A_1 \cup A_2| = |A_1| + |A_2| - |A_1 \cap A_2|\).}

\separate{7}\definition{14.1.3.1}{A \underline{derangement} of the numbers \(\{1,\;2,\;...,\;n\}\) is a permutation in which no number is in its correct position.}

\separate{7}\proposition{14.1.3.2}{The number of derangements of \(1,\;2,\;...,\;n\) is given by
\begin{equation}
    d(n) = \sum_{k=0}^{n} (-1)^k \frac{n!}{k!}
\end{equation}}
{We subtract from \(n!\) the number of non-derangements to get
\begin{equation}
    d(n) = n!\sum_{k=0}^{n} (-1)^k \frac{1}{k!}
\end{equation}}
In particular, the proportion of permutations that are derangements is
\begin{equation}
    \frac{d(n)}{n!} = n!\sum_{k=0}^{n} (-1)^k \frac{1}{k!}
\end{equation}
This is the power series expansion of \(e^{-1}\), so
\begin{equation}
    \lim_{n \rightarrow \infty} \frac{d(n)}{n!} = \frac{1}{e}
\end{equation}

\subsection{Recurrence Relations}

\definition{14.2.0.1}{As defined by \hyperref[cit:recurrenceRelation]{(Perepelitsa, 2020, p.1)}, a \underline{recurrence relation} for the sequence \(\{a_n\}\) is an equation that expresses \(a_n\) in terms of one or more of the previous terms of the sequence for all integers \(n\) with \(n \geq n_0\), where \(n_0\) is a non-negative integer.}

\separate{7}\definition{14.2.0.2}{As defined by \hyperref[cit:recurrenceRelation]{(Perepelitsa, 2020, p.1)}, a sequence is called a \underline{solution} of a recurrence relation if its terms satisfy the recurrence relation.}

\separate{7}\definition{14.2.0.3}{As defined by \hyperref[cit:recurrenceRelation]{(Perepelitsa, 2020, p.1)}, the \underline{initial conditions} for a sequence specify the terms that precede the first term where the recurrence relation takes effect.}

\separate{7}Let \(\{a_n\}\) be a sequence indexed by integers \(n = 0,\;1,\;2,\;...\), we are given:
\begin{itemize}
    \item The initial conditions \(a_0,\;a_1,\;...,\;a_{r-1}\).
    \item The recurrence relation 
    \begin{equation}\label{equ:14.2.0.1}
        a_n = \sum_{i=1}^{r} c_ia_{n-i} = c_1a_{n-1} + ... + c_ra_{n-r}, \quad (n \geq r)
    \end{equation}
    where \(c_1,\;...,\;c_r\) are specified constants.
\end{itemize}

\newpage\noindent\definition{14.2.0.4}{The recurrence relation is \underline{linear} because no term involves more than a single factor of the \(a_i\), and it is \underline{homogeneous} because every term involves an \(a_i\).}

\separate{7}\definition{14.2.0.5}{We seek solutions to the recurrence relation \eqref{equ:14.2.0.1} of the form \(a_n = x^n, \; x \neq 0\). Substituting this into the relation, we get 
\begin{equation}
    x^n = \sum_{i=1}^{r} c_ix^{n-i}, \quad(n \geq r)
\end{equation}
Dividing through by \(x^{n-r}\), we get
\begin{equation}
    x^r - \sum_{i=1}^{r} c_ix^{r-i} = 0
\end{equation}
This is the \underline{characteristic equation}.}
