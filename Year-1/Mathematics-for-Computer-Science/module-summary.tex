\part{Calculus}\label{part:1.calc}

\section{Introduction to Calculus}

\subsection{Sets}

\definition{1.1.1}{A \underline{set} is an unordered collection of distinct elements denoted as \(A = \{x | \textit{condition} \}\).}

\separate{5}Some common set operations include:
\begin{itemize}
    \item \definition{1.1.2}{The \underline{union} of two sets, \(A \cup B\), includes all the elements of both sets.}
    \item \definition{1.1.3}{The \underline{intersection} of two sets, \(A \cap B\), includes any elements which are present in both sets.}
    \item \definition{1.1.4}{If \(A\) and \(B\) are finite sets, then the \underline{principle of inclusion-exclusion} states that 
    \begin{equation}
        A \cup B = A + B - (A \cap B)
    \end{equation}}
    \item \definition{1.1.5}{The set \(A\) is a \underline{subset} of \(B\), \(A \subseteq B\), if every element of \(A\) is an element of \(B\). \(A\) is a \underline{proper subset} of \(B\), \(A \subset B\), if \(A\) is a subset of \(B\) and there is at least one element of \(B\) not in \(A\).}
    \item \definition{1.1.6}{The \underline{powerset} of a set \(A\) is the set of all subsets of \(A\). 
    \begin{equation}
        \mathcal{P}(A) = \{A'|A' \subseteq A\}
    \end{equation}}
    \item \definition{1.1.7}{The \underline{cartesian product} of two sets, \(A \times B\), is the set of all possible ordered pairs with the first element from \(A\) and the second from \(B\).}
\end{itemize}

\separate{5}\definition{1.1.8}{For a finite set, the \underline{cardinality} is the number of distinct elements that it contains.}

\subsection{Functions}

\separate{5}For a function \(f: A \rightarrow B\),
\begin{itemize}
    \item \definition{1.2.1}{\(A\) is known as the \underline{domain} of \(f\).}
    \item \definition{1.2.2}{\(B\) is known as the \underline{codomain} of \(f\).}
    \item \definition{1.2.3}{If \(f(a)=b\), then \(b\) is the \underline{image} of \(a\).}
    \item \definition{1.2.4}{The \underline{pre-image} of \(b\) is the set \(\{a|f(a)=b\}\).}
    \item \definition{1.2.5}{\(f\) forms an \underline{injection} if every element of \(A\) maps to a unique element in \(B\), even if some elements of \(B\) don't have a mapping.}
    \item \definition{1.2.6}{\(f\) forms a \underline{surjection} if every element of \(B\) maps to at least one element of \(A\).}
    \item \definition{1.2.7}{\(f\) forms a \underline{bijection} if it is both injective and surjective.}
\end{itemize}

\separate{5}\definition{1.2.8}{If for an unbounded set \(A\), there is a bijective function \(f: A \rightarrow \N\), then we say \(A\) is \underline{countably infinite}, otherwise \(A\) is known as \underline{uncountably infinite}.}

\subsection{Hyperbolic Functions}

\definition{1.3.1}{Given the unit hyperbola \(x^2-y^2=1\), the \underline{hyperbolic functions} are defined by the coordinates of a point at which area \(\frac{1}{2}\alpha\) has been swept out.}
\begin{equation*}
    sinh(\alpha) = \frac{e^\alpha - e^{-\alpha}}{2} \qquad cosh(\alpha) = \frac{e^\alpha + e^{-\alpha}}{2} \qquad tanh(\alpha) = \frac{e^x - e^{-x}}{e^x + e^{-x}} \qquad \sigma(\alpha) = \frac{1}{1 + e^{-\alpha}}
\end{equation*}
\begin{equation*}
    sech(\alpha) = \frac{2}{e^\alpha + e^{-\alpha}} \qquad cosech(\alpha) = \frac{2}{e^x - e^{-x}} \qquad coth(\alpha) = \frac{e^\alpha + e^{-\alpha}}{e^\alpha - e^{-\alpha}}
\end{equation*}
These functions can then be differentiated as follows:
\begin{equation*}
    \frac{d}{dx}\left[sinh(\alpha)\right] = cosh(\alpha) \qquad\qquad \frac{d}{dx}\left[cosh(\alpha)\right] = sinh(\alpha)
\end{equation*}
\begin{equation*}
    \frac{d}{dx}\left[tanh(\alpha)\right] = sech^2(\alpha) \qquad\qquad \frac{d}{dx}\left[\sigma(\alpha)\right] = \sigma(\alpha)(1-\sigma(\alpha))
\end{equation*}

\section{Multivariate Functions}

\subsection{Multivariate Derivatives}

\definition{2.1.1}{For a function \(f(x,y)\), if we take \(f=f_0\) to be fixed but unknown, then we can differentiate \(f\) as a function of \(x\) at \(x=x_0\), giving us the \underline{partial derivative of \(f\) with respect to \(x\)} which gives the gradient when moving in the \(x\)-direction from any point.}
\begin{equation}
    \frac{\delta f}{\delta x} \left[f(x,y)\right] = \lim_{x \rightarrow x_0} \left(\frac{f(x,y_0)-f(x_0,y_0)}{x-x_0}\right) = f_x
\end{equation}

\separate{5}\definition{2.1.2}{The \underline{vector of partial derivatives}, denoted as \(\nabla f\), contains all the partial derivatives of a function. This forms a vector pointing in the direction of the greatest rate of increase of \(f\), where its magnitude is the rate of increase in that direction. For example, for a function \(f(x,y)\), 
\begin{equation} 
    \nabla f = \begin{pmatrix}f_x\\f_y\end{pmatrix}
\end{equation}}

\separate{5}\definition{2.1.3}{The \underline{directional derivative} of the function \(f(x,y)\) gives the gradient at any point whilst looking in the direction given by the unit vector \(\mathbf{v} = \begin{pmatrix}v_x\\v_y\end{pmatrix}\).}
\begin{equation}
    \nabla_{\mathbf{v}} f = \lim_{h \rightarrow 0}\frac{f(x_0 + h\mathbf{v}) - f(x_0)}{h}
\end{equation}
If the slope is approximately flat at a small scale, then the slope can be determined by
\begin{equation}
    \nabla_{\mathbf{v}} f = \mathbf{v} \cdot \nabla f
\end{equation}

\separate{5}From this, we get that, for a function \(f(\mathbf{x_0})\) and a direction \(\mathbf{v}\),
\begin{equation}
    f(\mathbf{x_0} + h \mathbf{v}) \approx f(\mathbf{x_0}) + h \mathbf{v} \cdot \nabla f(\mathbf{x_0})
\end{equation}

\newpage\subsection{Multivariate Extrema}

\definition{2.2.1}{A function that is continuous is in \underline{class \(C^0\)}, and a function that is continuous and has a continuous derivative is in \underline{class \(C^1\)}. A function \(f\) such that \(f',\;f'',\;...,\;f^{(k)}\) exist and are continuous is in \underline{class \(C^k\)}.}

\separate{5}\definition{2.2.2}{A function \(f\) such that derivatives of all orders exist and are continuous is in class \(C^\infty\) is also \underline{smooth}.}

\separate{5}\definition{2.2.3}{All the second-order partial derivatives of a function \(f(x,y)\) can be gathered into a \underline{Hessian matrix}.}
\begin{equation}
    \hessian{f} = \begin{bmatrix} f_{xx} & f_{xy} \\ f_{yx} & f_{yy} \end{bmatrix}
\end{equation}
This can be used to determine the form of bivariate extrema. Suppose \(f(x,y) \in C^2\) and \(f_x(x_0,y_0) = 0 = f_y(x_0,y_0)\) then,
\begin{itemize}
    \item \((x_0,y_0)\) is a local maximum if \(det(\hessian{f}) > 0\) and \(f_{xx} < 0\) at \((x_0,y_0)\).
    \item \((x_0,y_0)\) is a local minimum if \(det(\hessian{f}) > 0\) and \(f_{xx} > 0\) at \((x_0,y_0)\).
    \item \((x_0,y_0)\) is a saddle point if \(det(\hessian{f}) < 0\).
    \item If \(det(\hessian{f}) = 0\), then the test is inconclusive.
\end{itemize}

\separate{5}At a stationary point \(x_0\), the quadratic approximation of a function \(f(x)\) is
\begin{equation}
    f(x_0 + \mathbf{v}) \approx f(x_0) + \frac{1}{2}\mathbf{v}^T H_f(x_0)\mathbf{v}
\end{equation}
But as \(\hessian{f}(x_0)\) is real and symmetric, it has orthogonal eigenvectors \(\mathbf{e_1},\;\mathbf{e_2},\;...,\;\mathbf{e_n}\) with eigenvalues \(\lambda_1,\;\lambda_2,\;...,\;\lambda_n\), so we can write
\begin{equation}
    \mathbf{v} = c_1\mathbf{e_1} + c_2\mathbf{e_2} + ... + c_n\mathbf{e_n}
\end{equation}
When substituted back into the original equation, this gives that
\begin{equation}
    f(x_0 + \mathbf{v}) \approx f(x_0) + \frac{1}{2}\left(\lambda_1(c_1)^2 + \lambda_2(c_2)^2 + ... + \lambda_n(c_n)^2\right)
\end{equation}
Given this, we can use the eigenvalues to determine the nature of the stationary point \(x_0\),
\begin{itemize}
    \item If all eigenvalues are positive, then the stationary point is a local minimum.
    \item If all eigenvalues are negaitve, then the stationary point is a local maximum.
    \item If there is a mixture of positive and negative eigenvalues, then the stationary point is a saddle point.
\end{itemize}

\subsection{Vector-Valued Functions}

\definition{2.3.1}{A \underline{vector-valued} function takes an \(n\)-dimensional vector as input, and outputs an \(m\)-dimensional vector, e.g., \(f:\R^n\rightarrow\R^m\).}

\separate{5}\definition{2.3.2}{The \underline{Jacobian matrix} of a vector-valued function \(f:\R^n\rightarrow\R^m\) can be formed by setting \(\jacobian{i,j} = \frac{\delta f_i}{\delta x_j}\) to form,}
\begin{equation}
    \jacobian{f} = \begin{pmatrix}
        \frac{\delta f_1}{\delta x_1} & \dots & \frac{\delta f_1}{\delta x_n} \\
        \vdots & \ddots & \vdots \\
        \frac{\delta f_m}{\delta x_1} & \dots & \frac{\delta f_m}{\delta x_n} \\
    \end{pmatrix}
\end{equation}
For a multivariate function \(f:\R^n\rightarrow\R^m\), the Jacobian matrix can be used to linearly approximate \(f\) at \(x\),
\begin{equation}
    f(x + h\mathbf{v}) \approx f(x) + \jacobian{f}(x)(h\mathbf{v})
\end{equation} 

\section{Lagrange Multipliers}

\subsection{Constrained Optimisation}

\definition{3.1.1}{The goal of a \underline{constrained optimisation} problem is to either minimise or maximise a multivariate function \(f(\mathbf{x})\) where \(x_1,\;x_2,\;...,\;x_n\) are not independent subject to some constraint \(g(\mathbf{x})\).}

\separate{5}\definition{3.1.2}{The \underline{Lagrange multipliers method} is used to find stationary points of an objective function \(f(\mathbf{x}) \in C^1\) subject to the constraint \(g(\mathbf{x}) = 0\).
\begin{enumerate}
    \item Define a new objective function called the \underline{Lagrangian},
    \begin{equation}
        \lagrange(\mathbf{x}, \lambda) = f(\mathbf{x}) - \lambda g(\mathbf{x})
    \end{equation}
    \item Find the stationary points of \(\lagrange\) with respect to both \(\mathbf{x}\) and \(\lambda\), i.e., solve the system of equations \(\nabla f(\mathbf{x}) = \lambda\nabla g(\mathbf{x})\) and \(g(\mathbf{x}) = 0\) for the unknowns \(\mathbf{x}\) and \(\lambda\).
    \item Then, the constrained extrema are found among the solutions to these equations. Identify their nature.
\end{enumerate}
}

\section{Series}

\subsection{Introduction to Series}

\definition{4.1.1}{A \underline{sequence} is an ordered, countably infinite, set of numbers.}

\separate{5}\definition{4.1.2}{A \underline{series} is the sum of the members of a sequence.}

\separate{5}\theorem{4.1.1}{If the series \(\sum^\infty_{n=0}a_n\) and the series \(\sum^\infty_{n=0}b_n\) are convergent, and have respective sums \(\alpha\) and \(\beta\), then
\begin{equation}
    \sum^\infty_{n=0} (a_n + b_n) = \alpha + \beta
\end{equation}
and 
\begin{equation}
    \sum^\infty_{n=0} (a_n - b_n) = \alpha - \beta
\end{equation}}
{\(\sum^\infty_{n=0}a_n\) converges to \(\alpha\) and \(\sum^\infty_{n=0}b_n\) converges to \(\beta\). Given an arbitrary \(\epsilon > 0\), let \(\epsilon_a = \frac{1}{2}\epsilon = \epsilon_b\). Since \(\sum^\infty_{n=0}a_n\) converges, \(\exists N_a\) such that,
\begin{equation}
    \left|\sum_{r=0}^{n}a_r - \alpha\right| < \epsilon_a,\;\forall n > N_a
\end{equation}
Similarly, since \(\sum^\infty_{n=0}b_n\) converges, \(\exists N_b\) such that,
\begin{equation}
    \left|\sum_{r=0}^{n}b_r - \beta\right| < \epsilon_b,\;\forall n > N_b
\end{equation}
\newpage\noindent Taking \(N = max\{N_a,N_b\}\), then \(\forall n > N\), 
\begin{equation}
    \left|\sum_{r=0}^{n}(a_r + b_r) - (\alpha + \beta)\right| = \left|\left(\sum_{r=0}^{n}a_r - \alpha\right) + \left(\sum_{r=0}^{n}b_r - \alpha\right)\right|
\end{equation}
\begin{equation*}
    \leq \left|\sum_{r=0}^{n}a_r - \alpha\right| + \left|\sum_{r=0}^{n}b_r - \beta\right| \leq \epsilon_a + \epsilon_b = \epsilon
\end{equation*}
i.e, \(\sum^\infty_{n=0}(a_n + b_n)\) converges to \((\alpha + \beta)\) using triangle inequality \(|a+b| \leq |a| + |b|\).
}

\separate{5}\theorem{4.1.2}{If \(\lim_{n \rightarrow \infty}a_n \neq 0\) then the series \(\sum_{n=0}^{\infty}a_n\) diverges.}
{Suppose \(\sum_{n=0}^{\infty}a_n\) converges for some \(S\), Then, \(\lim_{n\rightarrow\infty}S_n = S\), which means that given arbitrary \(epsilon > 0\), there exists some \(N_\epsilon\) such that, for all \(n > N_\epsilon\),
\begin{equation}
    |S_n - S| < \epsilon \qquad\qquad |S_n - S| < 2\epsilon
\end{equation}
Combining these, 
\begin{equation}
    2\epsilon > |S_n - S| + |S_{n+1} - S| = |S - S_n| + |S_{n+1} - S| \geq |S_{n+1} - S_n| = |a_{n+1}|
\end{equation}
But \(\epsilon\) was arbitrary, so we have shown that, for any \(\epsilon' > 0\) there exists an \(N_\epsilon\) such that \(|a_n| < \epsilon'\) for all \(n > N_\epsilon\). }

\separate{5}As the number of elements in a series approaches infinity...
\begin{itemize}
    \item \definition{4.1.3}{If \(\lim_{n\rightarrow\infty}a_n \neq 0\), then the series \underline{diverges}.}
    \item \definition{4.1.4}{If \(\lim_{n\rightarrow\infty}a_n = 0\), then the series \underline{converges}.}
    \item \definition{4.1.5}{If \(\lim_{n\rightarrow\infty}|a_n| = 0\), then the series \underline{absolutely converges}.}
    \item \definition{4.1.6}{If \(\lim_{n\rightarrow\infty}a_n = S\), then the series is \underline{convergent to \(S\)}.}
\end{itemize}

\separate{5}\theorem{4.1.3}{The \underline{ratio test} states that, if \(\sum^\infty_{n=1 a_n}\) is a series where \(a_n \neq 0\) and \(L = \lim_{n\rightarrow\infty \left|\frac{a_{n+1}}{a_n}\right|}\) then
\begin{enumerate}[label=\alph*)]
    \item If \(L < 1\), the series absolutely converges.
    \item If \(L > 1\), the series is divergent.
    \item If \(L =1 \), then the test fails.
\end{enumerate}
}{(a) If \(\lim_{n\rightarrow\infty}\left|\frac{a_{n+1}}{a_n}\right| = L < 1\), then \(\exists r\) with \(L < r < 1\) and \(N\) such that \(\forall n > N\):
\begin{equation}
    \lim_{n\rightarrow\infty}\left|\frac{a_{n+1}}{a_n}\right| < r \quad\rightarrow\quad |a_{n+1}| < r|a_n|
\end{equation}
Then \(|a_{n+2}| < r|a_{n+1}| < r^2|a_n|\) and \(|a_{n+3}| < r|a_{n+2} < r^3|a_n|\) etc. So the remainder is
\begin{equation}
    R_{n+m} = \lim_{m\rightarrow\infty}\left(|a_n|r^{m+1}\frac{1}{1-r}\right) = 0
\end{equation}
Hence the series converges.}

\separate{5}\definition{4.1.7}{The \underline{\(n^{th}\) root test} states that, for the series \(\sum^\infty_{n=1}a_n\), \(L=\lim_{n\rightarrow\infty}\sqrt[n]{|a_n|}\),
\begin{itemize}
    \item If \(L < 1\), then the series absolutely converges.
    \item If \(L > 1\), then the series is divergent.
    \item If \(L = 1\), then the test fails.
\end{itemize}}

\newpage\noindent\definition{4.1.8}{The \underline{alternating series test} states that the series \(\sum_{n=1}^{\infty}(-1)^{n+1}a_n\) converges if \(a_n > 0\), \(a_{n+1} \leq a_n\;\forall n\), and \(\lim_{n\rightarrow\infty}a_n = 0\)}

\separate{5}\theorem{4.1.4}{Every subsequence of a convergent sequence converges to the same limit as the original sequence.}
{Let \(\{a_{n_k}\}\) be a subsequence of \(\{a_n\}\). First note that \(n_k \geq k\) and so \(n_{k+1} \geq k + 1\). Let \(\lim_{n\rightarrow\infty}a_n = L\). Then, for any \(\epsilon > 0\), there exists \(N_\epsilon\) such that for all \(n > N_\epsilon\) it holds that \(|a_n-L|<\epsilon\). Let \(k > N_\epsilon\), then \(n_k > N_\epsilon\) and so it must be that \(|a_{n_k} - L| < \epsilon\).}

\subsection{Power Series}

\definition{4.2.1}{A \underline{power series} is a series involving a variable \(x\) and a constant \(x_0\) of the form
\begin{equation}
    \sum^\infty_{n=0}a_n(x-x_0)^n = a_0 + a_1(x-x_0) + a_2(x-x_0)^2 + ...
\end{equation}
At any specific value of \(x\), the power series becomes a normal series.}

\separate{5}\definition{4.2.2}{For an arbitrary power series \(\sum_{n=0}^{\infty}a_n(x-x_0)^n\), we can use the ratio test (including the \((x-x_0)\) term). From this, we call \(\lim_{n\rightarrow\infty}\left|\frac{a_n}{a_{n+1}}\right| = r\) the \underline{radius of convergence}. If \(|x-x_0| < r\), we have absolute convergence and if \(|x-x_0| > r\), we have divergence.}

\separate{5}Let \(f(x) = \sum_{n=0}^{\infty} a_nx^n\) be a power series with radius of convergence \(r > 0\). Then, within the interval \((-r,r)\):
\begin{itemize}
    \item \(f(x)\) is continuous.
    \item \(f'(x) = \sum_{n=1}^{\infty} na_nx^{n-1}\) (term-by-term differentiation).
    \item \(\int_{0}^{x} f(t) dt = \sum_{n=0}^{\infty} \frac{a_n}{n+1}x^{n+1}\) (term-by-term integration).
    \item Both series \(f'(x)\) and \(\int_{0}^{x}f(t) dt\) have the same radius of convergence as \(f(x)\).
\end{itemize}

\separate{5}\definition{4.2.3}{Suppose we start with some function \(f(x)\) and we wish to determine a power series \(\sum_{n=0}^{\infty}a_nx^n\) such that \(f(x)=\sum_{n=0}^{\infty}a_nx^n\). Putting these values in, we see that if such a power series exists, then:
\begin{equation}
    f(x) = f(0) + f'(0)x + f''(0)\frac{x^2}{2!} + f^{(3)}(0)\frac{x^3}{3!} + ... = \sum_{n=0}^{\infty} f^{(n)}(0)\frac{x^n}{n!}
\end{equation}
This is called the \underline{Maclaurin series for \(f\)}. To use the series, we must have that \(f \in C^\infty\) and the power series converges.}

\separate{5}\definition{4.2.4}{We can recenter the power series on \(x_0\) by defining a new function \(g(x) = f(x + x_0)\) then \(g(0) = f(x_0)\) and \(g'(x) = f'(x+x_0)\). Repeatedly differentiating, \(g^{(n)}(x) = f^{(n)}(x+x_0)\) so we build the Maclaurin series for \(g\), which gives us
\begin{equation}
    f(x+x_0) = g(x) = \sum_{n=0}^{\infty}g^{(n)}(0)\frac{x^n}{n!} = \sum_{n=0}^{\infty}f^{(n)}(0)\frac{x^n}{n!}
\end{equation}
or, 
\begin{equation}
    f(x) = g(x-x_0) = f(x_0) + f'(x_0)(x-x_0) + f''(x_0)\frac{(x-x_0)^2}{2!} + ...
\end{equation}
This form is called the \underline{Taylor series expansion of \(f\)}.}
