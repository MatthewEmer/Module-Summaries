\subsection{Probability and Statistics: Basic Concepts}

\subsubsection{Counting Theory}

\definition{The Product Rule}{The \underline{product rule} states that, if a procedure can be broken down into a sequence of two tasks with $n_1$ ways to do the first task and $n_2$ ways to do the second task, then there are $n_1n_2$ ways to complete the entire procedure.}

\separate{7}\example{Product Rule: Flights}
{John wants to fly to Canada from Germany with a connecting flight through the UK. There are 3 flights from Germany to the UK, and 2 flights from the UK to Canada. How many possible flight combinations are there?

\solution{ex:Product Rule: Flights}{\(3 \times 2 = 6\).}}

\separate{7}\definition{The Sum Rule}{The \underline{sum rule} states that, if you consider two independent sub-tasks (we only have to do one), if the first can be done in \(n_1\) ways, and the second can be done in \(n_2\) ways, then the total number of ways to complete the task is \(n_1+n_2\).}

\separate{7}\example{Sum Rule: Projects}
{Mary wants to choose a student project from three lists: lecturer 1 offers 5 projects, lecturer 2 offers 2 projects, and lecturer 3 offers 10 projects. How many possible options for her?

\solution{ex:Sum Rule: Projects}{\(5 + 2 + 10 = 17\).}}

\separate{7}\definition{The Inclusion-Exclusion Principle}{The \underline{inclusion-exclusion principle} states that, if two tasks can be done at the same time (unlike the sum rule in which they are independent), then there is a possibility of overcounting since some ways may be counted twice. To counter this, we add up the number of ways to do each task, then subtract the number of ways to do both tasks.
\begin{equation}
    |A \cup B| = \left(|A| + |B|\right) - \left(|A \cap B|\right)
\end{equation}
\begin{equation}
    |A \cup B \cup C| = \left(|A| + |B| + |C|\right) - \left(|A \cap B| + |A \cap C| + |B \cap C|\right) + \left(|A \cap B \cap C|\right)
\end{equation}}

\separate{7}\definition{r-permutations}{An \underline{r-permutation} of a set of \(r\) distinct objects is one ordered arrangement. We denote the number of r-permutations of a set with \(n\) elements in as \(P(n,r)\).
\begin{equation}\label{eq: permutations: distinct}
    P(n,r) = n \times (n-1) \times (n-2) \times ... \times (n-r+1) = \frac{n!}{(n-r)!}
\end{equation}

\noindent Suppose that a set consists of \(n\) objects, of which \(n_1\) are of one type (indistinguishable from one another), \(n_2\) are of a second type, ..., and \(n_k\) are of a \(k_{th}\) type where \(n = n_1 + n_2 + ... + n_k\). The number of permutations of the objects can be calculated with \eqref{eq: permutations: indistinguishable}.
\begin{equation}\label{eq: permutations: indistinguishable}
    P(n;n_1,n_2,...,n_k) = \frac{n!}{n_1!n_2!...n_k!}
\end{equation}} 

\separate{7}\definition{r-combinations}{A \underline{r-combination} is a counting of distinct objects in an unordered arrangement of length \(r\). The number of \(r\) combinations of a set with \(n\) distinct elements is given by \eqref{eq: combinations: distinct}.
\begin{equation}\label{eq: combinations: distinct}
    C(n,r) = \binom{n}{r} = \frac{n!}{(n-r)! \times r!}
\end{equation}}

\separate{7}\example{arrangements vs. combinations}{Assume elements \(A_1, A_2, A_3, A_4\) are in a set. We want to choose 2 elements from them.

\solution{ex:arrangements vs. combinations}{
\begin{itemize}
    \item  If the order of elements is important: \(P(4,2) = \frac{4!}{(4-2)!} = 12\).
    \item If the order of elements is not important: \(C(4,2) = \frac{4!}{(4-2)!2!} = 6\).
\end{itemize}}}

\subsubsection{Probability Theory}

\definition{Probability}{\underline{Probability} is the formalisation of uncertain events. It enables one to infer probabilities of interest based on assumptions and observations.}

\separate{7}\definition{Statistics}{\underline{Statistics} is the practice of the science of collecting and analysing data. It deals with experimental data and extracting knowledge from it.}

\separate{7}\definition{Experiment}{An \underline{experiment} is a procedure that yields one of a given set of possible outcomes.}

\separate{7}\definition{Sample Space}{The \underline{sample space} of the \hyperref[def:Experiment]{experiment} is the set of possible outcomes.}

\separate{7}\definition{Event}{An \underline{event} is any subset of the \hyperref[def:Sample Space]{sample space}.}

\separate{7}\theorem{Theory of Probability}{The \underline{Theory of Probability} states that, given a finite sample space of all outcomes \(S\), assuming that all outcomes are equally likely, for an event set \(E\) where \(E \subseteq S\), then the probability of occurrence of the event is given by \eqref{eq: Theory of Probability}.
\begin{equation}\label{eq: Theory of Probability}
    P(E) = \frac{|E|}{|S|}
\end{equation}}
{Proof omitted.}

\separate{7}\definition{Axioms of Probability}{The \underline{axioms of probability} are as follows:
\begin{itemize}
    \item \(P(E)\) is always between 0 and 1: \(0 \leq P(E) \leq 1\).
    \item \(P(S) = 1\).
    \item For disjoint events \(E_1,E_2,...,E_n\) (with no intersection), \(P(E_1 \cup E_2 \cup ... \cup E_n) = \sum_{i=1}^{n} P(E_i)\). 
\end{itemize}}
Then, from the \hyperref[def:Axioms of Probability]{Axioms of Probability} above, we can derive the following:
\begin{itemize}
    \item \definition{Complement Rule}{The \underline{complement rule} states that \(P(\overline{E}) = P(S-E) = P(S) - P(E) = 1 - P(E)\).}
    \item \definition{Impossible Rule}{The \underline{impossible rule} states that \(P(\emptyset) = 0\).}
    \item \definition{General Addition Rule}{The \underline{general addition rule} states that \(P(E_1 \cup E_2) = P(E_1) + P(E_2) - P(E_1 \cap E_2)\).}
    \item If \(E_1 \subseteq E_2\), then \(P(E_1) \leq P(E_2)\).
\end{itemize}

\separate{7}\example{Probability}{A sequence of 10 bits is randomly generated. What is the probability that at least 1 of these is 0?

\solution{ex:Probability}{Let \(E\) be an event in which all bits are 1.
\[P(E) = \frac{|E|}{|S|} = \frac{1}{2^{10}}\]
Then, by the \hyperref[def:Complement Rule]{complement rule}, \(P(\overline{E}) = P(1-E)\) where \(\overline{E}\) is the event in which at least one bit is not 1.
\[P(\overline{E}) = 1 - \frac{1}{1024} = \frac{1023}{1024}\]}}

\subsection{Dependency and Bayes Theorem}

\subsubsection{Conditional Probability}

The probability of event A, given that event B happens is given by:
\begin{equation}
    P(A|B) = \frac{P(A \cap B)}{P(B)}
\end{equation}
Therefore, the probability of both A and B occuring can be given by:
\begin{equation}
    P(A \cap B) = P(A|B) \times P(B)
\end{equation}

\separate{7}\definition{Independence}{Events can occur independently. The occurrence of one event is not dependent on the occurrence of another if, and only if:
\begin{equation}
    P(A \cap B) = P(A) \times P(B)
\end{equation}}

\separate{7}\definition{Bayes Theorem}{\(P(A \cap B)\) can be presented in two ways, either \(P(A|B) \times P(B)\) or \(P(B|A) \times P(A)\). Equating these, we can obtain \underline{Bayes' Theorem}.
\begin{equation}
    P(A|B) = \frac{P(B|A) \times P(A)}{P(B)}
\end{equation}
This can be rewritten by using variables, where \(P(A|B)\) is the posterior, \(P(A)\) is the prior, \(P(B|A)\) is the likelihood, and \(P(B)\) is the evidence.
\begin{equation}
    posterior = \frac{likelihood \times prior}{evidence}
\end{equation}
Suppose that \(E_1,E_2,...,E_n\) are mutually exclusive events such that:
\begin{equation}
    \bigcup^n_{i=1} E_i = S
\end{equation}
This means that one of the events \textbf{must} occur. Then, in any case, Bayes' Theorem states:
\begin{equation}
    P(E_k|A) = \frac{P(E_k) \times P(A|E_k)}{\sum_{j=1}^{n} P(E_j) \times P(A|E_j)}
\end{equation}}

\subsection{Discrete Probability Distributions}

\subsubsection{Random Variables}

\definition{Random variables}{A \underline{random variable} is a function that assigns a number to each outcome in the sample space of a random experiment. 
It is also known as a \underline{stochastic variable}. 
They are usually denoted by a capital \(X\) or \(Y\).}

\separate{7}\definition{Discrete random variables}{We say that \(X\) is a \underline{discrete random variable} if \(X\) can take a finite number of \(k\) values \(x_1,.x_2,...,x_k\) and an infinite sequence of different values \(x_1,x_2,...\) which can be distinguished from one another.}

\separate{7}\definition{Probability distribution}{The probability distribution of a random variable \(X\) describes the probabilities associated with each possible outcome of \(X\).
Let \(X\) be a discrete random variable. 
Suppose that the possible values it can assume are \(x_1,x_2,...,x_n\).
Then, these values are associated with probabilities given by
\begin{equation}\label{equation:discrete distribution}
    P(X = x_k) = f(x_k), k=1,2,...,n
\end{equation}
This function \(f\) is known as the probability mass function (PMF).}

\subsubsection{Distribution Functions}

\separate{7}\definition{PMF}{The \underline{probability mass function (PMF)} is function where the value is a probability and has the following characteristics:
\begin{itemize}
    \item \(f(x_k) \geq 0\).
    \item \(\sum_{k=1}^{n} f(x_k) = 1\).
\end{itemize}
The mathematical definition is that, for a discrete random variable \(X\) with possible values \(x_1,x_2,...\) the probability mass function is a function \(f:\{x_1,x_2,...\} \rightarrow [0,1]\) satisfying \(f(x_i) \geq 0\), \(\sum_{k=1}^{\infty} f(x_i) = 1\), and \(P(X = x_i) = f(x_i)\).
They can be calculated by collecting all possible outcomes where \(X=x\) and summing their probabilities for all \(x\).}

\separate{7}\definition{CDF}{The \underline{cumulative distribution function (CDF)} or \underline{distribution function} for a random variable \(X\) is defined as
\begin{equation}
    F(x) = P(X \leq x) = \sum_{x_i \leq x} P(X = x_i) = \sum_{x_i \leq x} f(x_i), \text{ where } x \in \R
\end{equation}}

\separate{7}\definition{Mean}{The \underline{mean (\(\mu\)) or expected value \(E\)} of a function is the measure of dispersion or variability in a distribution.}

\separate{7}\definition{Variance}{The \underline{variance} of a function is the measure of dispersion or variability in a distribution.}

\separate{7}It is important to consider two laws when discussing random variables:
\begin{enumerate}
    \item Functions of random variables are also random variables. 
        Let \(X\) be a random variable, and \(h:\R\rightarrow\R\) be an arbitrary function.
        Then \(Y = h(X)\) is also a random variable.
    \item \definition{Law of the Unconscious Statistician}{The \underline{law of the unconscious statistician}: Let \(X\) be a random variable with PMF \(f\) and \(h:\R\rightarrow\R\) be an arbitrary function.
        Then \(Y=h(X)\) is also a random variable. If \(X\) is discrete and takes values \(\{x_1,x_2,...\}\) then
        \begin{equation}
            E[Y] = E[h(x)] = \sum_i h(x_i)f(x_i)
        \end{equation}}
\end{enumerate}
These two laws result in the conclusion that one can perform arithmetic operations with random variables to get new ones, and the expectation and variance can be computed from them.
\begin{itemize}
    \item \(E(aX + b) = aE[X] + b\).
    \item \(E[h(X)] = \sum_x h(x)f(x)\).
    \item \(V(aX + b) = a^2V(X)\).
    \item \(V(X) = E(X^2) - [E(X)]^2\).
\end{itemize}
From this, \(E[X]\) is named the first moment, and \(E[X]^2\) is the second moment. The variance is also called the central second moment.

\separate{7}We can consider two random variables \(X\) and \(Y\) to be independent if, for sets \(A\) and \(B\), we have
\begin{equation}
    P(\{X \in A\} \cap \{Y \in B\}) \equiv P(X \in A, Y \in B) = P(X \in A)P(Y \in B)
\end{equation}
For independent random variables only, we have
\begin{equation}
    V(X_1 + X_2 + ... + X_n) = V(X_1) + V(X_2) + ... + V(X_n)
\end{equation}
This equation holds for expectation \(E\), whether the variables are independent or not.

\subsubsection{Special Discrete Distributions}

\definition{Discrete Uniform Distribution}{The \underline{discrete uniform distribution} is the simplest discrete random variable.
It assumes a finite number of possible values, each with equal probability.
A random variable \(X\) has a discrete uniform distribution if each of the \(n\) values in its range \(x_1,x_2,...,x_n\) has equal probability.
Suppose random variable \(X\) has uniform distribution. 
The probability mass function of \(X\) is given by (\(X \sim Uni(n)\)).
\begin{equation}
    P(X = k) = f(k) = \frac{1}{n}
\end{equation}
If we assume \(X\) is a discete uniform distribution on the consecutive integers \(a, a+1, a+2,...,b\) for \(a < b\), then:
\begin{equation}
    \mu = E[X] = \frac{a+b}{2}
\end{equation}
\begin{equation}
    \sigma^2 = Var(X) = \frac{(b-a+1)^2-1}{12}
\end{equation}}

\separate{7}\definition{Bernoulli Distribution}{The most common discrete distributions are associated with \underline{Bernoulli trials}.
In a random experiment if each outcome consists of only two probabilities, such as in a toss of a coin, then the random experiment is called a Bernoulli trial.
In a Bernoulli trial, let the possible events in each outcome be denoted by \(A\) and \(B\).
Then, \(A \cap B = \emptyset\) and \(A \cup B = S\).
Let the occurrence of \(A\) be a success, and the occurrence of \(B\) be a fail and let the probability of \(A\) be \(p\), then:
\begin{equation}\label{equation:Bernoulli q}
    P(B) = P(\overline{A}) = 1 - P(A) = 1 - p = q
\end{equation}
Let \(X\) be a random variable taking only two possible values \(\{0,1\}\) and let \(p = P(X=1)\).
Then \(X\) is said to be a \underline{Bernoulli random variable} with parameter \(p\): \(X \sim Ber(p)\), then:
\begin{equation}
    E[X] = 0 \times P(X = 0) + 1 \times P(X = 1) = p
\end{equation}
\begin{equation}
    E[X^2] = 0^2 \times P(X=0) + 1^2 \times P(X=1) = p
\end{equation}
\begin{equation}\label{equation:Bernoulli V}
    V(X) = E[X^2] - (E[X])^2 = p-p^2 = p(1-p)
\end{equation}
Taking \eqref{equation:Bernoulli q} and \eqref{equation:Bernoulli V}, we get that \(E[X] = p\) and \(V(X) = pq\).}

\separate{7}\definition{Discrete Binomial Distribution}{Consider a random experiment consisting of \(n \in \N\) independent Bernoulli trials.
If the probability of success for each trial equals \(p\), \(0 \leq p \leq 1\), and it remains the same throughout the experiment, then we say that the random variable \(X\) that equals the number of successful trials is a \underline{binomial random variable} with parameters \(n\) and \(p\).
The probability mass function of \(X\) is given by \(X \sim Bin(n,p)\).
\begin{equation}
    P(X = k) = f(k) = \binom{n}{k}p^k(1-p)^{n-k}, k=0,1,...,n
\end{equation}
The binomial random variable is \(n\) independent repetitions of a Bernoulli variable, \(X = X_1 + X_2 + ... + X_n\). 
So we can calculate 
\begin{equation}
    E[X] = E[X_1 + X_2 + ... + E_n] = E[X_1] + E[X_2] + ... + E[X_n]
\end{equation}
\begin{equation}
    V(X) = V[X_1 + X_2 + ... + X_n] = V(X_1) + V(X_2) + ... + V(X_n)
\end{equation}
Therefore, \(E[X] = np\) and \(V(X) = np(1-p)\).}

\separate{7}\definition{Discrete Geometric Distribution}{Consider a series of independent Bernoulli trials with constant probability of success \(p\).
Let the random variable \(X\) denote the number of trials until the first success. 
Then \(X\) is called a \underline{geometric random variable} with parameter \(p\): \(X \sim Geom(p)\).
\begin{equation}
    P(X = k) = f(k) = p(1-p)^{k-1}, k=1,2,...
\end{equation}
and \(E[X] = \frac{1}{p}\) and \(V(X) = \frac{1-p}{p^2}\).}

\separate{7}\theorem{The Memoryless Property}{The geometric distribution has no memory of the past. 
This is called the \underline{memoryless} \underline{property}, meaning that if you have waited \(k\) trials already without seeing a success, you must wait \(m\) more trials to get success with probability of \(p(1-p)^{m-1}\): \(P(X = k+m | X > k) = P(X =m)\).}
{\begin{equation}
    P(X = k + m | X > k) = \frac{P(\{X = k+m\} \cap \{X > k\})}{P(\{X > k\})}
\end{equation}
\begin{equation}
    \frac{P(X = k+m)}{P(X > k)} = \frac{P(X = k+m)}{1 - P(X \geq k)}
\end{equation}
\begin{equation}
    = \frac{p(1-p)^{k+m-1}}{(1-p)^k} = p(1-p)^{m-1} = P(X = m)
\end{equation}}

\separate{7}\definition{Negative Binomial Distribution}{We can generalise the idea behind the geometric distribution - instead of waiting for the first success, we can wait until \(r\) successes.
Consider a series of independent Bernoulli trials with constant success probability \(p\). 
Let the random variable \(X\) denote the number of trials until \(r \in \N\) successes are attained.
Then \(X\) is called a \underline{negative binomial random variable} with parameter \(p\) and \(r\): \(X \sim NegBin(p,r)\).
\begin{equation}
    P(X=k) = f(k) = \begin{cases}
                        0, \text{ if } k < r \\
                        \binom{k-1}{r-1}p^r(1-p)^{k-r}, \text{ if } k \geq r
                    \end{cases}, \text{ where } k \in \N
\end{equation}
Then \(E[X] = \frac{r}{p}\) and \(V(X) = r \frac{1-p}{p^2}\).}

\separate{7}\definition{Poisson}{Suppose that you want to model the number of customers entering a store over the period of one day.
To do this we make two assumptions: the customers behave independently from one another and the expected number of customers per unit of time is constant \(\lambda\).

Breaking the day into four-hour intervals gives six pieces \((n=6)\), so the expected number of customers for each interval is \(\frac{\lambda}{n} = \frac{\lambda}{6}\).
If the intervals are really small, then the number of customers is either 0 or 1 with very high probability. 
Then the number of customers in each interval is approximately a Bernoulli random variable with \(p = \frac{\lambda}{n}\), they are also all independent.

Now look at getting \(k\) arrivals in a \(Bin(n,p)\), where the number of bins \(n\) grows towards \(+\infty\).
In this case, the probability of a single event will be \(p = \frac{\lambda}{n}\).

A \underline{Poisson random variable} has a distribution that shows how many times an event is likely to happen within a specific period of time.
\(X\) is called a \underline{Poisson random variable} with parameter \(\lambda > 0\). 
It is the number of events in a Poisson process, \(X \sim Poisson(\lambda)\).
\begin{equation}
    P(X=k) = f(k) = \frac{e^{-\lambda}\lambda^k}{k!}, k=0,1,2,...
\end{equation}
and \(E[X] = \lambda\) and \(V(X) = \lambda\).}

\subsection{Continuous Probability Distributions}

\subsubsection{Continuous Random Variables}

\definition{Continuous Random Variable}{A \underline{continuous random variable} is a random variable such that the probability of any one specific value is unknown, and thus is treated as zero.
Instead, we calculate the probability of the value being in a given interval.}

\separate{7}\definition{PDF}{In general, continuous random variables are described by a \underline{probability density function}. 
It maps a probability of a range of events from the sample space to a function.
The probability density function of a continuous random variable \(X\) is denoted \(f_X\) or simply \(f\).
\begin{equation}
    P(a \leq X \leq b) = \int_{a}^{b} f_X (x) dx
\end{equation}
We define the probability density function \(f(x)\) for a continuous random variable \(X\) such that:
\begin{itemize}
    \item \(f(x) \geq 0, \forall x \in \R\).
    \item \(\int_{-\infty}^{+\infty} f(x) dx = 1\).
    \item \(P(a \leq X \leq b) = \int_{a}^{b} f(x) dx, \forall (a,b) | (a,b) \in \R^2, a \leq b\).
\end{itemize}}

\separate{7}\definition{Continuous CDF}{Similarly, we can define the \underline{continuous cumulative distribution function} \(F(x):\R \rightarrow [0,1]\) of a continuous random variable \(X\) where \(x \in \R\) as
\begin{equation}
    F(x) = P(X \leq x) = \int_{-\infty}^{x} f(t) dt
\end{equation}
As in the discrete case, a continuous CDF has the following properties:
\begin{itemize}
    \item \(F(x)\) is non-decreasing: if \(x \leq y\) then \(F(x) \leq F(y)\).
    \item \(0 \leq F(x) \leq 1\): \(\lim_{x \rightarrow -\infty} F(x) = 0\), \(\lim_{x \rightarrow +\infty} F(x) = 1\).
    \item For \(a \leq b\) we have: \(P(a \leq X \leq b) = F(b) - F(a)\).
\end{itemize}}

\separate{7}If \(F(x)\) is the cumulative distribution function of a continuous random variable \(X\), then its derivative is the probability distribution function \(f(x)\).
\begin{equation}
    f(x) = F'(x) = \frac{d}{dx}F(x) 
\end{equation}

\separate{7}\example{continuous random variables question}{A continuous random variable \(X\) has the density function \eqref{equation:example:continuous random variables question}.
Find the value of the constant \(c\).
\begin{equation}\label{equation:example:continuous random variables question}
    f(x) = \frac{c}{x^2+1} \text{ where } -\infty < x < +\infty 
\end{equation}

\separate{3}\solution{example:continuous random variables question}
{From \postenvref{def:PDF}{Definition}{definitionnumber}, we know that 
\begin{equation}
    F(+\infty) = P(X \leq +\infty) = \int_{-\infty}^{+\infty} f(t) dt = 1
\end{equation}
Substituting in \(f(x)\), we get
\begin{equation}
    \int_{-\infty}^{+\infty} \frac{c}{x^2+1} dx = \left[\arctan(x)\right]_{-\infty}^{+\infty} = 1
\end{equation}
\begin{equation}

\end{equation}
This gives us that, \(1 = c\pi\), giving \(c = \frac{1}{\pi}\).}}

\subsubsection{Mean and Variance}

\separate{7}Going back to the continuous random variable \(X\), we can define its mean and variance as values to summarise its distribution such that,
\begin{equation}
    \mu = E[X] = \int_{-\infty}^{+\infty} x f(x) dx
\end{equation}
and 
\begin{equation}
    \sigma^2 = V(X) = E\left[(X - \mu)^2\right] = \int_{-\infty}^{+\infty} (x-\mu)^2 f(x) dx
\end{equation}
As for discrete random variables, mean and variance have the following properties:
\begin{itemize}
    \item \(E[aX + b] = aE[X] + b\).
    \item \(V(aX + b) = a^2V(X)\).
    \item \(\sqrt{V(aX+b)} = |a|\sqrt{V(X)}\).
\end{itemize}

\subsubsection{Specific Continuous Distributions}

\definition{Continuous Uniform Distribution}{The \underline{continuous uniform distribution} sets all samples in a range as having the same probability.
Formally, let \(X\) be a continuous random variable with density function
\begin{equation}
    f(x) = \begin{cases}
                \frac{1}{b-a}, \text{ if } a \leq x \leq b \\
                0, \text{ otherwise}
            \end{cases}
\end{equation}
We have that the mean and variance of this distribution are \(E(X) = \frac{b+a}{2}\) and \(V(X) = \frac{(b-a)^2}{12}\).}

\separate{7}\definition{Exponential Distribution}{The \underline{exponential distribution} is used to model the time elapsing between events.
The probability of occurrence of an event after a unit of time.
It has a parameter \(\lambda\) which is the rate parameter and is the average number of events occuring in a period of one time unit.

Formally, a random variable \(X\) is called an \underline{exponential random variable} with parameter \(\lambda > 0\) (denoted by \(X \sim Exp(\lambda)\)) when the probability density function is
\begin{equation}
    f(x) = \begin{cases}
                \lambda e^{-\lambda x}, \text{ if } x \geq 0 \\
                0, \text{ otherwise}
            \end{cases}
\end{equation}
and 
\begin{equation}
    F(x) = \begin{cases}
                1 - e^{-\lambda x}, \text{ if } x \geq 0 \\
                0, \textit{ otherwise}
            \end{cases}
\end{equation}
We can then define the mean and variance as \(E(X) = \frac{1}{\lambda}\) and \(V(X) = \frac{1}{\lambda^2}\).}

\subsection{Data Quality and How to Produce High-Quality Data}

\subsubsection{Data Quality}

\definition{Data Cleansing}{\underline{Data cleansing} is the process of detecting and correcting (or removing) corrupt or inaccurate records from a record set, table, or database and refers to identifying incomplete, inaccurate, or irrelevant parts of the data and then replacing, modifying, or deleting the dirty or coarse data.}

\separate{7}For data to be considered high-quality, it should meet the following criteria:
\begin{itemize}
    \item \definition{Data Quality: Validity}{\underline{Validity}: The degree to which the measures conform to defined business rules or constraints.}
    \item \definition{Data Quality: Accuracy}{\underline{Accuracy}: The degree of conformity of a measure to a standard or a true value.}
    \item \definition{Data Quality: Completeness}{\underline{Completeness}: The degree to which all required measures are known.}
    \item \definition{Data Quality: Consistency}{\underline{Consistency}: The degree to which the data is consistent, within the same data set or across multiple data sets.}
    \item \definition{Data Quality: Uniformity}{\underline{Uniformity}: The degree to which the data is specified using the same unit of measure.}
\end{itemize}

\subsubsection{How to Produce High-Quality Data}

Producing high-quality data is an iterative process moving through \hyperref[def:Inspection]{inspection}, \hyperref[def:Cleaning]{cleaning}, \hyperref[def:Verifying]{verifying}, and \hyperref[def:Reporting]{reporting}.

\separate{7}\definition{Inspection}{\underline{Inspection}: Detect unexpected, incorrect, and inconsistent data.}
\begin{itemize}
    \item \definition{Data Profiling}{\underline{Data profiling} is the process of viewing the summary statistics of your data to identify standards or patterns in columns of data. Here it is also important to note the data type, any missing values, the number of unique values, the distribution of values, and the relationship between the data and other columns.}
    \item Visualising the data can also be helpful for identifying outliers, as can using statistical methods such as mean, standard deviation, range, and/or quartiles.
\end{itemize}

\separate{7}\definition{Cleaning}{\underline{Cleaning}: Fix or remove the anomalies detected.}
\begin{itemize}
    \item It is important to identify and remove duplicate entries in your data sets.
    \item When converting the data into a consistent data type, values unable to fit the format should be flagged and converted to NA value.
    \item Beyond the data types, the data should be further standardised. This means strings are of a consistent capitalisation, numerical values use the same measurement unit, and dates are in the same format (pick either the UK or US format).
    \item It can also be useful to scale/transform your data to fit within a certain scale (i.e. 0-100 or 0-1). This can make data easier to plot (e.g., exponential data being logged).
    \item You can either replace missing values by using a mask to indicate where they are missing, or choosing a sentinel value to replace the missing entries. Conversely, you can also drop the whole record from the data set, or where possible calculate the missing values from other observations.
    \item \definition{Outliers}{We define \underline{outliers} as values which fall 1.5 times the interquartile range from both the first and third quartiles.}
\end{itemize}

\separate{7}\definition{Verifying}{\underline{Verifying}: After cleaning, the results are inspected to verify correctness.}

\separate{7}\definition{Reporting}{\underline{Reporting}: A report about the changes made and the quality of the currently stored data is recorded.}
